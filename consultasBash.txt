ForEach-Object { $_.ToString() } |
ForEach-Object -Begin {$file=""} -Process {
    if ($_ -match "\.py") { $file = $_ }
    elseif ($_ -match "LOC:\s+(\d+)") {
        [PSCustomObject]@{
            File = $file
            LOC = [int]$matches[1]
        }
    }
} | Sort-Object LOC -Descending
