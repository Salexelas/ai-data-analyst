def normalize_query_for_intent(text: str):
    text = normalize(text)  # ya tenés esto

    replacements = {
        "más": "mas",
        "top": "top",
        "mayores": "top",
        "mejores": "top",

        "menos": "bottom"
    }

    tokens = text.split()
    tokens = [replacements.get(t, t) for t in tokens]

    return " ".join(tokens)


def fuzzy_match_score(q, term):
    q_tokens = set(q.split())
    t_tokens = set(term.split())

    if not t_tokens:
        return 0

    overlap = len(q_tokens & t_tokens)
    return overlap / len(t_tokens)

def normalize(text):
    text = text.lower()
    text = unicodedata.normalize("NFKD", text)
    text = "".join(c for c in text if not unicodedata.combining(c))
    text = re.sub(r"[^a-z0-9\s]", "", text)
    return text


def is_match(q, term):
    return term in q or q in term


def token_overlap(a, b):
    a_tokens = set(a.split())
    b_tokens = set(b.split())
    return len(a_tokens & b_tokens) > 0


def is_valid_question_basic(question: str):
    q = question.strip().lower()

    if len(q) < 3:
        return False

    if not any(c.isalpha() for c in q):
        return False

    return True


def get_search_terms(semantic_layer):

    terms = list(
        semantic_layer.get("metrics", {}).keys()
    )

    for aliases in semantic_layer.get(
        "aliases",
        {},
    ).values():

        if isinstance(aliases, list):
            terms.extend(aliases)
        else:
            terms.append(aliases)

    terms.extend(
        semantic_layer.get("columns", [])
    )

    return [
        normalize(term)
        for term in terms
    ]


def match_score(question, term):

    question_tokens = set(question.split())
    term_tokens = set(term.split())

    if not term_tokens:
        return 0

    coverage = (
        len(question_tokens & term_tokens)
        / len(term_tokens)
    )

    substring = int(term in question)

    return coverage + 0.3 * substring


def is_question_related_fast(
    question,
    semantic_layer,
):

    if not semantic_layer:
        return True

    question = normalize(question)

    return any(
        match_score(question, term) >= 0.5
        for term in get_search_terms(
            semantic_layer
        )
    )


def detect_language_fast(text):
    text = text.lower()

    scores = {
        "es": 0,
        "en": 0,
        "pt": 0,
        "de": 0,
        "fr": 0,
        "it": 0
    }

    lang_keywords = {
        "es": ["el", "la", "los", "las", "de", "que", "en", "y", "para", "con", "por", "estadisticas", "equipo"],
        "en": ["the", "is", "are", "of", "and", "in", "for", "with", "stats", "team"],
        "pt": ["o", "a", "os", "as", "de", "que", "em", "para", "com", "estatisticas", "equipe"],
        "de": ["der", "die", "das", "und", "ist", "in", "für", "mit", "statistik", "mannschaft"],
        "fr": ["le", "la", "les", "de", "et", "est", "dans", "pour", "avec", "statistiques", "equipe"],
        "it": ["il", "lo", "la", "gli", "le", "di", "e", "è", "per", "con", "statistiche", "squadra"]
    }

    tokens = text.split()

    for lang, keywords in lang_keywords.items():
        for token in tokens:
            if token in keywords:
                scores[lang] += 1

    best_lang = max(scores, key=scores.get)

    if scores[best_lang] == 0:
        return "en"

    return best_lang

INTENT_KEYWORDS = {
    "top": ["top", "mas", "mayores"],
    "bottom": ["bottom", "menos", "peores"],
    "average": ["promedio", "average"],
    "total": ["total", "suma"],

UNKNOWN_INTENT = {
    "intent": "unknown",
    "metric": None,
    "dimension": None,
    "limit": None,
}


def normalize_intent(intent: dict) -> dict:
    return {
        "intent": intent.get("intent"),
        "metric": intent.get("metric"),
        "dimension": intent.get("dimension"),
        "limit": int(intent.get("limit")) if intent.get("limit") else None,
    }


def detect_fast_intent(query: str) -> str | None:
    for intent, keywords in INTENT_KEYWORDS.items():
        if any(keyword in query for keyword in keywords):
            return intent
    return None


def detect_limit(query: str) -> int | None:
    match = re.search(r"\b(\d+)\b", query)
    return int(match.group(1)) if match else None


def build_llm_prompt(question: str) -> str:
    return f"""
Classify the user intent.

Return ONLY JSON:

{{
  "intent": "top | bottom | average | total | comparison | distribution | unknown",
  "metric": "string or null",
  "dimension": "string or null",
  "limit": "number or null"
}}

Question:
{question}
"""


def detect_llm_intent(question: str, llm) -> dict:
    raw = llm.generate(
        build_llm_prompt(question),
        system_prompt="Intent classifier",
    )

    try:
        return normalize_intent(json.loads(raw))
    except (json.JSONDecodeError, TypeError, ValueError):
        return UNKNOWN_INTENT.copy()


def detect_intent(question: str, llm, semantic_layer=None) -> dict:
    query = normalize_query_for_intent(question)

    intent = detect_fast_intent(query)
    limit = detect_limit(query)

    metric = detect_metric(query, semantic_layer)
    dimension = detect_dimension(query, semantic_layer)

    if intent:
        return normalize_intent(
            {
                "intent": intent,
                "metric": metric,
                "dimension": dimension,
                "limit": limit,
            }
        )

    return detect_llm_intent(question, llm)

    dataset_ids,
    dataset_relations=None,
    semantic_layer=None,
    llm=None,

    try:

        ctx = build_context(
            question,

            dataset_relations,
            semantic_layer,
            llm,
            trace_id,

            ctx,
            llm,
        )

            ctx,

        store_response(
            ctx,
            response,
        )

        return response

    except Exception as e:

        return ErrorHandler.to_response(e)

IMPORTANT:

- If relation is one_to_many or many_to_many, aggregate after JOIN.

    return f"""
Return ONLY valid JSON.

Format:
{{
  "sql": "string",
  "confidence": float

Rules:
- Only SELECT
- Use ONLY columns from schema
- No explanation
- No markdown

- total -> SUM()

Available tables:
{available_tables}

{intent_text}

Schema:
{schema.to_dict(orient="records")}

{semantic_text}

{relationships_text}

def validate_llm_output(parsed):

    if not isinstance(parsed, dict):
        raise LLMOutputFormatError("Output is not a JSON object")

    if "sql" not in parsed:
        raise LLMOutputFormatError("Missing 'sql' field")

    if not isinstance(parsed["sql"], str):
        raise LLMOutputFormatError("SQL must be string")


def generate_sql_structured(
    question,
    schema,
    semantic_layer,
    llm,
    language="en",
    intent=None,
    relationships=None
):

    prompt = build_prompt(
        question=question,
        schema=schema,
        semantic_layer=semantic_layer,
        intent=intent,
        language=language,
        relationships=relationships
    )

    raw = llm.generate(
        prompt,
        system_prompt="You are a senior data analyst. Return ONLY valid JSON."
    )

    if not raw or not raw.strip():
        raise LLMOutputFormatError("Invalid JSON: empty response")

    raw = raw.strip()
    raw = raw.replace("```json", "").replace("```", "").strip()

    try:
        decoder = json.JSONDecoder()
        parsed, _ = decoder.raw_decode(raw)

    except Exception:
        print("LLM RAW RESPONSE:")
        print(raw)
        raise LLMOutputFormatError(f"Invalid JSON: {raw}")

    validate_llm_output(parsed)

    return parsed

from sqlglot import exp

from core.settings import SQLConfig


def guard_sql(
    query: str,
    schema_columns: list[str],
    allowed_tables: list[str] | None = None,
) -> str:
    """
    Validate a SQL query before execution.
    """

    if not query:
        raise ValueError("Empty query")

    try:
        parsed = sqlglot.parse_one(query)
    except Exception as exc:
        raise ValueError("Invalid SQL") from exc

    if not isinstance(parsed, exp.Select):
        raise ValueError("Only SELECT statements are allowed")

    _validate_tables(parsed, allowed_tables)
    _validate_columns(parsed, schema_columns)
    _validate_joins(parsed)

    if not parsed.args.get("limit"):
        parsed.set(
            "limit",
            exp.Limit(
                expression=exp.Literal.number(SQLConfig.MAX_ROWS)
            ),
        )

    return parsed.sql()


def _validate_tables(
    parsed: exp.Expression,
    allowed_tables: list[str] | None,
) -> None:

    if not allowed_tables:
        return

    used_tables = {
        table.name.lower()
        for table in parsed.find_all(exp.Table)
    }

    invalid_tables = (
        used_tables
        - {table.lower() for table in allowed_tables}
    )

    if invalid_tables:
        raise ValueError(
            f"Invalid tables: {sorted(invalid_tables)}"
        )


def _validate_columns(
    parsed: exp.Expression,
    schema_columns: list[str],
) -> None:

    used_columns = {
        column.name
        for column in parsed.find_all(exp.Column)
    }

    invalid_columns = (
        used_columns
        - set(schema_columns)
    )

    if invalid_columns:
        raise ValueError(
            f"Invalid columns: {sorted(invalid_columns)}"
        )


def _validate_joins(
    parsed: exp.Expression,
) -> None:

    joins = list(parsed.find_all(exp.Join))

    if len(joins) > SQLConfig.MAX_JOINS:
        raise ValueError("Too many JOINs")

    if any(
        join.args.get("kind") == "cross"
        for join in joins
    ):
        raise ValueError(
            "CROSS JOIN is not allowed"
        )

def should_retry(error: str):

    error = error.lower()

    retryable = [
        "invalid json",
        "missing 'sql'",
        "sql inválido",
        "columnas inválidas",
        "tabla inválida",
        "solo select",
        "demasiados joins",
        "expecting value",
        "invalid json: empty response",
    ]

    return any(
        x in error
        for x in retryable
    )


def generate_sql_with_retry(
    question,
    schema,
    semantic_layer,
    columns,
    llm,
    language="en",
    intent=None,
    relationships=None,
    max_retries=2,
):

    last_error = None
    previous_output = None

    for _ in range(
        max_retries + 1
    ):

        try:

            if previous_output is None:

                parsed = (
                    generate_sql_structured(
                        question=question,
                        schema=schema,
                        semantic_layer=semantic_layer,
                        llm=llm,
                        language=language,
                        intent=intent,
                        relationships=relationships,
                    )
                )

            else:

                parsed = (
                    repair_sql_generation(
                        question=question,
                        schema=schema,
                        semantic_layer=semantic_layer,
                        intent=intent,
                        relationships=relationships,
                        previous_output=previous_output,
                        error=last_error,
                        llm=llm,
                    )
                )

            return validate_sql(
                parsed,
                schema,
                columns,
            )

        except Exception as e:

            last_error = str(e)
            previous_output = parsed

            if not should_retry(
                last_error
            ):
                raise

    raise ValueError(
        f"Failed after retries: {last_error}"
    )

        raise ValueError(
            "SQL inválido"
        )

        raise ValueError(
            "SQL demasiado corto"
        )

        schema["table_name"]
        .unique()
        .tolist()
    )

    return guard_sql(
        sql,
        columns,
        allowed_tables=tables,
    )

def normalize_table_name(file_name: str):
    name = (
        file_name
        .replace(".csv", "")
        .replace(".xlsx", "")
        .lower()
        .replace(" ", "_")
        .replace("-", "_")
    )
    return re.sub(r"[^a-z0-9_]", "_", name)


RELATION_TYPES = {
    (True, True): "one_to_one",
    (True, False): "one_to_many",
    (False, True): "many_to_one",
    (False, False): "many_to_many",
}


def infer_relation_type(left_df, right_df, left_col, right_col):

    key = (
        left_df[left_col].is_unique,
        right_df[right_col].is_unique,
    )

    return RELATION_TYPES[key]

class MultiDatasetConnector:

    def __init__(self, dataset_ids, relations=None):

        self.conn = duckdb.connect()
        self.schemas = []
        self.tables = {}
        self.relations = relations or []

        for file_name in dataset_ids:

            path = os.path.join("data", file_name)

            if file_name.endswith(".csv"):
                df = pd.read_csv(path)

            elif file_name.endswith(".xlsx"):
                df = pd.read_excel(path)

            else:
                continue

            table_name = normalize_table_name(file_name)

            self.tables[file_name] = {
                "table_name": table_name,
                "df": df
            }

            self.conn.register(table_name, df)

            schema = self.conn.execute(
                f"DESCRIBE {table_name}"
            ).df()

            schema["table_name"] = table_name
            self.schemas.append(schema)

        self._enrich_relations()

    def _enrich_relations(self):

        enriched = []

        for rel in self.relations:

            left_df = self.tables[rel["left_dataset"]]["df"]
            right_df = self.tables[rel["right_dataset"]]["df"]

            cardinality = infer_relation_type(
                left_df,
                right_df,
                rel["left_column"],
                rel["right_column"]
            )

            rel["cardinality"] = cardinality

            rel["left_table"] = self.tables[
                rel["left_dataset"]
            ]["table_name"]

            rel["right_table"] = self.tables[
                rel["right_dataset"]
            ]["table_name"]

            enriched.append(rel)

        self.relations = enriched

    def execute(self, query: str):
        return self.conn.execute(query).df()

    def get_schema(self):
        return pd.concat(self.schemas, ignore_index=True)

    def get_relationships(self):
        return self.relations

def call_llm(prompt: str, system_prompt: str = "You are a helpful AI") -> str:

    payload = {
        "model": LLMConfig.MODEL,
        "messages": [
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": prompt}
        ],
        "temperature": LLMConfig.TEMPERATURE
    }

    start = time.time()

    response = requests.post(
        API_URL,
        headers=headers,
        json=payload,
        timeout=LLMConfig.TIMEOUT
    )

    duration = time.time() - start
    print(f"🔥 LLM RESPONSE TIME: {duration:.2f}s")

    if response.status_code != 200:
        raise Exception(response.text)

    result = response.json()

    return result["choices"][0]["message"]["content"]

class Step:

class Pipeline:

    def __init__(self, steps):
        self.steps = steps

    def run(self, ctx):
        for step in self.steps:
            start = time.time()

            log_event(
                "step_start",
                ctx.trace_id,
                step=step.name
            )

            try:
                step.run(ctx)

                duration = time.time() - start

                log_event(
                    "step_success",
                    ctx.trace_id,
                    step=step.name,
                    duration=duration
                )

            except Exception as e:
                duration = time.time() - start

                log_event(
                    "step_error",
                    ctx.trace_id,
                    step=step.name,
                    duration=duration,
                    error=str(e)
                )

                raise

        return ctx

        if file_path.endswith(".csv"):

def load_dataset(service, file_path):

    try:
        return service.load_preview(file_path, limit=1000)

    except Exception as e:
        st.error(f"Error cargando dataset: {e}")
        return None


def render_tabs(service, df, schema):

    tabs = st.tabs([
        "📝 Columnas",
        "🧠 Tipos",
        "📊 Datos"
    ])

    with tabs[0]:
        rename_map = render_columns_tab(df)

    with tabs[1]:
        column_types = render_types_tab(service, df, schema)

    typed_df = service.apply_types(df, column_types)

    with tabs[2]:
        edited_df = render_data_tab(typed_df)

    edited_df.columns = [
        rename_map[c]
        for c in edited_df.columns
    ]

    return rename_map, column_types, edited_df


def render_columns_tab(df):

    rename_map = {}
    cols = st.columns(2)

    for i, col in enumerate(df.columns):

        with cols[i % 2]:
            rename_map[col] = st.text_input(
                col,
                value=col,
                key=f"rename_{col}"
            )

    return rename_map


def render_types_tab(service, df, schema):

    column_types = {}
    cols = st.columns(2)

    for i, col in enumerate(df.columns):

        with cols[i % 2]:

            saved_type = schema.get(col)

            default_type = (
                saved_type
                if saved_type
                else service.detect_type(df[col])
            )

            column_types[col] = st.selectbox(
                f"Tipo → {col}",
                TYPE_OPTIONS,
                index=TYPE_OPTIONS.index(default_type),
                key=f"type_{col}"
            )

    return column_types


def render_data_tab(df):

    llm = st.session_state.get("llm")

        st.session_state["edited_df"] = df.copy()

  dataset_editor\services\transform_service.py
  import pandas as pd
import json
import re


class TransformService:

    def clean_json(self, response):

        response = response.strip()
        response = response.replace("```json", "")
        response = response.replace("```", "")

        match = re.search(
            r"\[.*\]",
            response,
            re.DOTALL
        )

        if match:
            return match.group(0)

        return response

    def _transform_values(
        self,
        values,
        column,
        instruction,
        llm
    ):

        prompt = f"""
Transformá valores de una columna.

COLUMNA:
{column}

VALORES:
{values}

INSTRUCCIÓN:
{instruction}

REGLAS:
- aplicar la instrucción a cada valor
- devolver SOLO JSON array
- misma cantidad de elementos
- mismo orden
- sin explicación
"""

        response = llm.generate(
            prompt=prompt,
            system_prompt="data transformer"
        )

        cleaned = self.clean_json(response)

        return json.loads(cleaned)

    def preview_transform(
        self,
        df,
        column,
        instruction,
        llm
    ):

        sample = (
            df[column]
            .dropna()
            .astype(str)
            .head(20)
            .tolist()
        )

        try:

            transformed = self._transform_values(
                sample,
                column,
                instruction,
                llm
            )

            return pd.DataFrame({
                "original": sample,
                "transformado": transformed
            })

        except Exception as e:

            return pd.DataFrame({
                "error": [str(e)]
            })

    def apply_transform(
        self,
        df,
        column,
        instruction,
        llm
    ):

        df = df.copy()

        values = (
            df[column]
            .dropna()
            .astype(str)
            .unique()
            .tolist()
        )

        transformed = self._transform_values(
            values,
            column,
            instruction,
            llm
        )

        mapping = dict(zip(values, transformed))

        df[column] = (
            df[column]
            .astype(str)
            .replace(mapping)
        )

        return df

  import pandas as pd


TYPE_OPTIONS = [
    "auto",
    "text",
    "number",
    "datetime",
    "boolean",
    "category",
    "id"
]


class TypeService:

    DETECTION_RULES = [
        ("datetime", "datetime"),
        ("bool", "boolean"),
        ("int", "number"),
        ("float", "number"),
    ]

    CONVERTERS = {
        "datetime": (
            pd.to_datetime,
            {
                "dayfirst": True,
                "errors": "coerce",
            },
        ),
        "number": (
            pd.to_numeric,
            {
                "errors": "coerce",
            },
        ),
        "category": (
            pd.Series.astype,
            {
                "dtype": "category",
            },
        ),
        "text": (
            pd.Series.astype,
            {
                "dtype": "string",
            },
        ),
    }

    @staticmethod
    def detect(series):

        dtype = str(series.dtype)

        for pattern, detected_type in TypeService.DETECTION_RULES:

            if pattern in dtype:
                return detected_type

        return "text"

    @staticmethod
    def _to_boolean(series):

        return (
            series
            .astype("string")
            .str.strip()
            .str.lower()
            .map({
                "true": True,
                "false": False,
                "1": True,
                "0": False,
                "yes": True,
                "no": False,
            })
        )

@staticmethod
def apply(df, schema):

    df = df.copy()

    for col, dtype in schema.items():

        if col not in df.columns:
            continue

        TypeService._apply_column(df, col, dtype)

    return df


@staticmethod
def _apply_column(df, col, dtype):

    try:

        if dtype == "boolean":
            df[col] = TypeService._to_boolean(df[col])
            return

        converter = TypeService.CONVERTERS.get(dtype)

        if not converter:
            return

        func, kwargs = converter

        df[col] = func(df[col], **kwargs)

    except (ValueError, TypeError):
        pass

                    "Probá reformular la pregunta."
                ),
            )

        return AnalyticsResponse(
            type="error",
            data=[],
            markdown=f"Error: {exception}",
        )

class InsightService:

    def __init__(self, llm):
        self.llm = llm

    def generate(self, ctx):

        try:

            df = ctx.output.result

            if df is None or len(df) == 0:

                ctx.output.markdown = (
                    "# ⚠️ No data"
                )

                ctx.output.metrics = {}
                ctx.output.chart_type = None
                ctx.output.severity = "warning"

                return

            df = df.round(2)

            preview = (
                df.head(10)
                .to_dict(
                    orient="records"
                )
            )

            data = {
                "rows": preview,
                "intent":
                    ctx.exec.intent.get(
                        "intent"
                    )
                    if ctx.exec.intent
                    else None
            }

            prompt = build_insight_prompt(
                ctx.input.question,
                data,
                ctx.exec.language
            )

            response = (
                self.llm.generate(
                    prompt,
                    system_prompt=
                    "You are a senior BI analyst. Return ONLY valid JSON."
                )
            )

            parsed = (
                parse_analytics_response(
                    response
                )
            )

            ctx.output.markdown = (
                parsed.get(
                    "markdown"
                )
            )

            ctx.output.chart_type = (
                parsed.get(
                    "chart_type"
                )
            )

            ctx.output.metrics = (
                parsed.get(
                    "metrics",
                    {}
                )
            )

            ctx.output.severity = (
                parsed.get(
                    "severity",
                    "info"
                )
            )

        except Exception as e:

            ctx.output.markdown = (
                f"# ❌ Insight Error\n\n{e}"
            )

            ctx.output.metrics = {}
            ctx.output.chart_type = None
            ctx.output.severity = "error"

def render_analytics(response):

    # -------------------------
    # EXECUTIVE MARKDOWN
    # -------------------------
    if response.markdown:

        st.markdown(
            response.markdown
        )

    # -------------------------
    # METRICS
    # -------------------------
    if response.metrics:

        st.markdown("## 📌 KPIs")

        cols = st.columns(
            len(response.metrics)
        )

        for col, (k, v) in zip(
            cols,
            response.metrics.items()
        ):

            col.metric(k, v)

    # -------------------------
    # DATAFRAME
    # -------------------------
    if response.data is not None:

        st.markdown("## 📊 Datos")

        st.dataframe(
            response.data
        )

    # -------------------------
    # CHARTS
    # -------------------------
    if response.chart_type == "line":

        st.line_chart(
            response.data
        )

    elif response.chart_type == "bar":

        st.bar_chart(
            response.data
        )

    # -------------------------
    # SQL
    # -------------------------
    if response.sql:

        with st.expander(
            "🛠 SQL"
        ):

            st.code(
                response.sql,
                language="sql"
            )

def extract_json(text: str):
    match = re.search(r"\{.*\}", text, re.DOTALL)
    if match:
        return match.group(0)
    return "{}"

Return ONLY valid JSON.

    try:

  semantic\semantic_layer.py
  def build_semantic_layer(
    columns,
    aliases=None,
    metrics=None,
    dimensions=None,
):
    return {
        "columns": columns,
        "aliases": aliases or {},
        "metrics": metrics or {},
        "dimensions": dimensions or {},
    }


def normalize_aliases(values):
    if values is None:
        return []

    if isinstance(values, str):
        return [values]

    return values


def aliases_match(query, aliases):
    return any(
        alias in query
        for alias in normalize_aliases(aliases)
    )


def find_by_alias(query, items):
    for name, config in items.items():
        candidates = [name] + normalize_aliases(
            config.get("aliases")
        )

        if aliases_match(query, candidates):
            return name

    return None


def detect_metric(query, semantic_layer):
    if not semantic_layer:
        return None

    return find_by_alias(
        query,
        semantic_layer.get("metrics", {}),
    )


def detect_dimension(query, semantic_layer):
    if not semantic_layer:
        return None

    dimension = find_by_alias(
        query,
        semantic_layer.get("dimensions", {}),
    )

    if dimension:
        return dimension

    aliases = {
        column: {
            "aliases": normalize_aliases(values)
        }
        for column, values in semantic_layer.get(
            "aliases", {}
        ).items()
    }

    return find_by_alias(query, aliases)


def merge_aliases(base_aliases, extra_aliases):
    merged = base_aliases.copy()

    for column, values in extra_aliases.items():
        existing = normalize_aliases(
            merged.get(column)
        )

        merged[column] = list(
            dict.fromkeys(
                existing
                + normalize_aliases(values)
            )
        )

    return merged


def merge_semantic_layers(base_layer, extra_layer):
    if not base_layer:
        return extra_layer

    if not extra_layer:
        return base_layer

    return {
        "columns": list(
            dict.fromkeys(
                base_layer.get("columns", [])
                + extra_layer.get("columns", [])
            )
        ),
        "aliases": merge_aliases(
            base_layer.get("aliases", {}),
            extra_layer.get("aliases", {}),
        ),
        "metrics": {
            **base_layer.get("metrics", {}),
            **extra_layer.get("metrics", {}),
        },
        "dimensions": {
            **base_layer.get("dimensions", {}),
            **extra_layer.get("dimensions", {}),
        },
    }

  import io
import easyocr
import numpy as np

from PIL import Image, ImageEnhance, ImageFilter


class OCRService:

    def __init__(self):

        self.reader = easyocr.Reader(
            ["es", "en"],
            gpu=False
        )

    def extract_text(self, file_bytes):

        image = Image.open(
            io.BytesIO(file_bytes)
        )

        image = image.convert("L")
        image = image.filter(ImageFilter.SHARPEN)
        image = ImageEnhance.Contrast(image).enhance(2.5)

        image_np = np.array(image)

        results = self.reader.readtext(
            image_np,
            detail=1,
            paragraph=False
        )

        rows = []

        for result in results:

            bbox, text, conf = result

            if conf < 0.25:
                continue

            x = bbox[0][0]
            y = bbox[0][1]

            rows.append({
                "x": x,
                "y": y,
                "text": text
            })

        # ordenar visualmente
        rows = sorted(
            rows,
            key=lambda r: (round(r["y"] / 20), r["x"])
        )

        return "\n".join(
            [
                r["text"]
                for r in rows
            ]
        )

class OCRLLMParserService:

    def clean_json(self, response):

        response = response.strip()
        response = response.replace("```json", "")
        response = response.replace("```", "")

        match = re.search(
            r"\[.*\]",
            response,
            re.DOTALL
        )

        if match:
            return match.group(0)

        return response

    def parse_to_dataframe(
        self,
        raw_text,
        columns,
        examples,
        context,
        llm
    ):

        if not columns:
            return pd.DataFrame({
                "error": ["No columns"]
            })

        prompt = f"""
Extraé dataset estructurado.

CONTEXTO:
{context}

TEXTO OCR:
{raw_text}

COLUMNAS:
{", ".join(columns)}

EJEMPLOS:
{examples}

REGLAS:
- devolver SOLO JSON
- no inventar
- null si falta
"""

        response = llm.generate(
            prompt=prompt,
            system_prompt="OCR parser"
        )

        cleaned = self.clean_json(response)

        try:
            return pd.DataFrame(json.loads(cleaned))

        except Exception:
            return pd.DataFrame({
                "raw_output": [response]
            })

from paddleocr import PaddleOCR


class OCRService:

    def __init__(self):

        self.ocr = PaddleOCR(
            use_angle_cls=True,
            lang="en"
        )

    def extract_text(self, file_bytes):

        with tempfile.NamedTemporaryFile(
            delete=False,
            suffix=".png"
        ) as f:

            f.write(file_bytes)
            temp_path = f.name

        result = self.ocr.ocr(temp_path)

        lines = []

        for page in result:
            for item in page:

                text = item[1][0]
                score = item[1][1]

                if score > 0.4:
                    lines.append(text)

        return "\n".join(lines)

  from faster_whisper import WhisperModel
import tempfile
import os


class WhisperService:

    def __init__(self):

        self.model = WhisperModel(
            "base",
            device="cpu",
            compute_type="int8"
        )

    def transcribe(self, audio_bytes):

        with tempfile.NamedTemporaryFile(
            delete=False,
            suffix=".wav"
        ) as f:

            f.write(audio_bytes)
            audio_path = f.name

        segments, info = self.model.transcribe(
            audio_path
        )

        text = " ".join(
            segment.text
            for segment in segments
        )

        os.remove(audio_path)

        return text.strip()

class InsightTool:

    def __init__(self, llm):
        self.llm = llm

    def run(self, question, context):

        if "language" not in context:
            raise ValueError("Language not provided in context")

        df = context.get("last_result")

        if df is None or df.empty:
            return "No hay datos para analizar."

        lang = context["language"]

        preview = df.head(20).to_string()

        prompt = f"""
You are analyzing query results.

Question:
{question}

Data:
{preview}

Instructions:
- Provide a concise business insight
- Do not mention SQL
- Focus on key patterns, trends, or anomalies

IMPORTANT:
- The response MUST be written in language: {lang}
- Do NOT use any other language
"""

        return self.llm.generate(
            prompt,
            system_prompt=f"""
You are a senior business data analyst.

CRITICAL RULE:
- You ALWAYS respond in language: {lang}
- You NEVER switch language
"""
        )

    )

    if st.button("Process image"):

        with st.spinner("Reading image..."):
            st.session_state["ocr_text"] = OCRPipeline.extract(
                uploaded_file
            )

    text = st.session_state.get("ocr_text")

    if not text:
        return

    st.text_area(
        "Detected text",
        value=text,
        height=250
    )

    columns = st.text_input(

    )

    generate_dataset(

        columns
    )

def generate_dataset(
    text,
    columns
):

    if not st.button("Generate dataset"):
        return

    st.session_state["ocr_df"] = OCRPipeline.build_dataframe(
        raw_text=text,
        columns=columns,
        context="",
        examples="",
        llm=st.session_state["llm"]
    )


def preview_dataset():

    df = st.session_state.get("ocr_df")

    if df is None:
        return

    edited_df = st.data_editor(
        df,
        width="stretch",
        num_rows="dynamic"
    )

    errors = OCRPipeline.validate(
        edited_df
    )

    if errors:
        for error in errors:
            st.error(error)
        return

    st.success("Dataset valid ✔")

    if not st.button("Save dataset"):
        return

    result = OCRPipeline.save(
        edited_df,
        prefix="ocr_simple"
    )

    st.success(result["dataset_id"])
    st.rerun()

  def apply_aggregation(
    df,
    x_col,
    y_col,
    aggregation,
    group_col
):

    # ==================================================
    # NO AGG
    # ==================================================
    if aggregation == "none":

        return df, y_col

    # ==================================================
    # COUNT
    # ==================================================
    if aggregation == "count":

        if group_col != "Sin grupo":

            result = (
                df
                .groupby([x_col, group_col])
                .size()
                .reset_index(name="count")
            )

        else:

            result = (
                df
                .groupby(x_col)
                .size()
                .reset_index(name="count")
            )

        return result, "count"

    # ==================================================
    # OTHER AGGS
    # ==================================================
    if group_col != "Sin grupo":

        result = (
            df
            .groupby([x_col, group_col])[y_col]
            .agg(aggregation)
            .reset_index()
        )

    else:

        result = (
            df
            .groupby(x_col)[y_col]
            .agg(aggregation)
            .reset_index()
        )

    return result, y_col

def render_filters(df):

    columns = list(df.columns)

    st.subheader("🎯 Filtros")

    filter_col = st.selectbox(
        "Filtrar por",
        ["Sin filtro"] + columns
    )

    if filter_col == "Sin filtro":

        return None

    unique_values = (
        df[filter_col]
        .dropna()
        .unique()
        .tolist()
    )

    selected_values = st.multiselect(
        "Valores",
        unique_values,
        default=unique_values[:5]
        if len(unique_values) > 5
        else unique_values
    )

    return {
        "column": filter_col,
        "values": selected_values
    }


def apply_filters(df, filters):

    if not filters:

        return df

    return df[
        df[filters["column"]]
        .isin(filters["values"])
    ]

  import streamlit as st
import plotly.express as px

from ui.components.charts.chart_aggregations import apply_aggregation
from ui.components.charts.chart_filters import apply_filters


CHART_BUILDERS = {
    "line": px.line,
    "scatter": px.scatter,
    "bar": px.bar,
}


def unpack_config(config):
    return (
        config["chart_type"],
        config["x"],
        config["y"],
        config["aggregation"],
        config["group_by"],
        config["filters"],
    )


def prepare_chart_data(
    df,
    x_col,
    y_col,
    aggregation,
    group_col,
    filters,
):
    df = apply_filters(df, filters)

    if str(df[x_col].dtype).startswith("datetime"):
        df = df.sort_values(x_col)

    return apply_aggregation(
        df,
        x_col,
        y_col,
        aggregation,
        group_col,
    )


def build_chart(
    chart_type,
    chart_df,
    x_col,
    y_axis,
    group_col,
):
    if chart_type == "pie":
        return px.pie(
            chart_df,
            names=x_col,
            values=y_axis,
        )

    builder = CHART_BUILDERS.get(chart_type)

    if builder is None:
        raise ValueError(f"Unsupported chart type: {chart_type}")

    color = None if group_col == "Sin grupo" else group_col

    return builder(
        chart_df,
        x=x_col,
        y=y_axis,
        color=color,
    )


def render_chart_output(fig, chart_df):
    st.plotly_chart(

    )

def render_general_tab(df):

    col1, col2, col3, col4 = st.columns(4)

    col1.metric("Rows (sample)", len(df))
    col2.metric("Columns", len(df.columns))
    col3.metric("Missing", int(df.isna().sum().sum()))

    memory_mb = round(
        df.memory_usage(deep=True).sum() / 1024 / 1024,
        2
    )

    col4.metric("Memory MB", memory_mb)


def render_schema_tab(df):

    schema_df = pd.DataFrame({
        "column": df.columns,
        "dtype": df.dtypes.astype(str),
        "nulls": df.isna().sum().values,
        "unique": df.nunique().values
    })

    st.dataframe(schema_df, width="stretch")


def render_numeric_tab(numeric_df):

    if numeric_df.empty:
        st.info("No hay columnas numéricas")
        return

    stats = pd.DataFrame({
        "mean": numeric_df.mean(),
        "median": numeric_df.median(),
        "std": numeric_df.std(),
        "min": numeric_df.min(),
        "max": numeric_df.max(),
        "missing": numeric_df.isna().sum()
    })

    st.dataframe(
        stats.round(2),
        width="stretch"
    )

    st.bar_chart(numeric_df.mean())


def render_categorical_tab(cat_df):

    if cat_df.empty:
        st.info("No hay columnas categóricas")
        return

    for col in cat_df.columns[:5]:

        st.subheader(col)

        st.bar_chart(
            cat_df[col]
            .astype(str)
            .value_counts()
            .head(10)
        )


def render_correlation_tab(numeric_df):

    if len(numeric_df.columns) <= 1:
        st.info("No hay suficientes columnas numéricas")
        return

    corr = numeric_df.corr(numeric_only=True)

    st.dataframe(
        corr.round(2),
        width="stretch"
    )


def render_raw_tab(df):

    st.dataframe(
        df.head(100),
        width="stretch"
    )

import time
import hashlib
import json


class SimpleCache:
    def __init__(self, ttl=300):
        self.store = {}
        self.ttl = ttl

    def get(self, key):
        item = self.store.get(key)

        if not item:
            return None

        if time.time() - item["ts"] > self.ttl:
            del self.store[key]
            return None

        return item["value"]

    def set(self, key, value):
        self.store[key] = {
            "value": value,
            "ts": time.time()
        }


def build_cache_key(question, dataset_id, semantic_layer, intent, language):
    base = {
        "q": question,  # 🔥 IMPORTANTE: ahora sí incluimos la query
        "dataset": dataset_id,
        "semantic": semantic_layer,
        "intent": intent,
        "lang": language
    }

    raw = json.dumps(base, sort_keys=True, default=str)
    return hashlib.md5(raw.encode()).hexdigest()

slack_bot.py
import os

from dotenv import load_dotenv

from slack_bolt import App
from slack_bolt.adapter.socket_mode import SocketModeHandler

from core.llm_service import LLMService
from agents.service import answer_question

from connectors.factory import create_connector

# =========================
# LOAD ENV
# =========================

load_dotenv()

# =========================
# SLACK
# =========================

app = App(
    token=os.getenv("SLACK_BOT_TOKEN"),
    signing_secret=os.getenv("SLACK_SIGNING_SECRET")
)

# =========================
# LLM
# =========================

llm = LLMService()

# =========================
# DATASET
# =========================

DATA_PATH = "data"

files = sorted(
    os.listdir(DATA_PATH),
    reverse=True
)

last_file = files[0]

file_path = os.path.join(
    DATA_PATH,
    last_file
)

if last_file.endswith(".csv"):

    file_type = "csv"

elif last_file.endswith(".xlsx"):

    file_type = "excel"

else:

    raise ValueError(
        "Unsupported dataset"
    )

connector = create_connector(
    file_type,
    {"file_path": file_path}
)

# =========================
# EVENTS
# =========================

@app.event("message")
def handle_message(body, say):

    event = body["event"]

    # -------------------------
    # IGNORE BOTS
    # -------------------------
    if event.get("bot_id"):

        return

    question = event.get(
        "text",
        ""
    )

    try:

        result = answer_question(

            question=question,

            connector=connector,

            dataset_id=last_file,

            semantic_layer=None,

            llm=llm
        )

        # -------------------------
        # ERROR
        # -------------------------
        if result.type == "error":

            response = getattr(
                result,
                "message",
                "Error"
            )

        # -------------------------
        # ANALYTICS
        # -------------------------
        else:

            markdown = (
                result.markdown
                or ""
            )

            data = result.data

            preview = ""

            if data is not None:

                preview = (
                    data.head(10)
                    .to_string(index=False)
                )

            response = f"""
{markdown}

```{preview}```
"""

    except Exception as e:

        response = f"❌ Error: {e}"

    say(response)

# =========================
# START
# =========================

if __name__ == "__main__":

    handler = SocketModeHandler(
        app,
        os.getenv("SLACK_APP_TOKEN")
    )

    handler.start()
