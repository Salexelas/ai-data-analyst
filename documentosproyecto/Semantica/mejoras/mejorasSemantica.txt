import os
import json


BASE_PATH = "semantic_store"


def save_semantic_layer(dataset_id, semantic_layer):
    os.makedirs(BASE_PATH, exist_ok=True)

    path = f"{BASE_PATH}/{dataset_id}.json"

    with open(path, "w") as f:
        json.dump(semantic_layer, f, indent=2)


def load_semantic_layer(dataset_id):
    path = f"{BASE_PATH}/{dataset_id}.json"

    if not os.path.exists(path):
        return None

    with open(path) as f:
        return json.load(f)
🧠 3. Generar dataset_id inteligente

👉 NO uses nombre de archivo
👉 usá hash del schema

ejemplo:
import hashlib

def generate_dataset_id(schema_df):
    schema_str = "|".join(schema_df["column_name"].tolist())
    return hashlib.md5(schema_str.encode()).hexdigest()
🚀 4. Integrarlo en router.py
from semantic_store import load_semantic_layer, save_semantic_layer
from utils import generate_dataset_id


def route_question(question, connector, semantic_layer=None, dataset_id=None):

    schema = connector.get_schema()

    # 🔥 generar dataset_id automático si no viene
    if not dataset_id:
        dataset_id = generate_dataset_id(schema)

    # 🔥 cargar semántica existente
    stored_semantic = load_semantic_layer(dataset_id)

    # 🔥 merge semántico (clave)
    if stored_semantic and semantic_layer:
        stored_semantic["aliases"].update(semantic_layer.get("aliases", {}))
        stored_semantic["metrics"].update(semantic_layer.get("metrics", {}))
        semantic_layer = stored_semantic

        # guardar actualización
        save_semantic_layer(dataset_id, semantic_layer)

    elif stored_semantic:
        semantic_layer = stored_semantic

    elif semantic_layer:
        save_semantic_layer(dataset_id, semantic_layer)

    # 🔥 generar SQL con semántica persistente
    sql = generate_sql(question, schema, semantic_layer)

    try:
        result = connector.execute(sql)

        if dataset_id:
            save_query(dataset_id, question, sql)

        return result

    except Exception as e:
        return {"error": str(e)}
