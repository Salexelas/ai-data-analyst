import re

def clean_sql(output: str) -> str:
    match = re.search(r"(SELECT|WITH).*", output, re.IGNORECASE | re.DOTALL)
    return match.group(0).strip() if match else output.strip()

🧱 5. Cómo implementarla YA (simple pero potente)
Paso 1: crear estructura
semantic_layer = {
    "aliases": {},
    "metrics": {}
}
Paso 2: UI en Streamlit (CLAVE)

Después de mostrar columnas:

st.subheader("🧠 Capa Semántica")

aliases = {}
for col in preview.columns:
    alias = st.text_input(f"Alias para {col}", col)
    aliases[col] = alias

st.session_state["semantic_layer"] = {
    "aliases": aliases
}
Paso 3: usarla en el prompt

En generate_sql:

def generate_sql(question, schema, semantic_layer=None):

    semantic_text = ""
    if semantic_layer:
        semantic_text = f"Aliases: {semantic_layer['aliases']}"

    prompt = f"""
    You are a senior SQL expert.

    ONLY return SQL.

    Schema:
    {schema}

    {semantic_text}

    Question:
    {question}

    SQL:
    """
Paso 4: pasarla desde router
def route_question(question, connector, semantic_layer=None):
    schema = connector.get_schema()
    sql = generate_sql(question, schema, semantic_layer)

Y en main:

route_question(
    question,
    st.session_state["connector"],
    st.session_state.get("semantic_layer")
)
