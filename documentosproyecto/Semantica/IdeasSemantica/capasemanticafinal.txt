import hashlib

def get_dataset_id(source_type, config):
    base = f"{source_type}_{str(config)}"
    return hashlib.md5(base.encode()).hexdigest()
Guardarlo en session:

import json
import os

def save_semantic_layer(dataset_id, semantic_layer):
    os.makedirs("semantic_store", exist_ok=True)

    path = f"semantic_store/{dataset_id}.json"

    with open(path, "w") as f:
        json.dump(semantic_layer, f, indent=2)

def load_semantic_layer(dataset_id):
    path = f"semantic_store/{dataset_id}.json"

    if os.path.exists(path):
        with open(path) as f:
            return json.load(f)

    return None
Después de conectar:

Guardar queries
def save_query(dataset_id, question, sql):
    os.makedirs("query_history", exist_ok=True)

    path = f"query_history/{dataset_id}.json"

    data = []

    if os.path.exists(path):
        with open(path) as f:
            data = json.load(f)

    data.append({
        "question": question,
        "sql": sql
    })

    with open(path, "w") as f:
        json.dump(data, f, indent=2)

⭐ 7. PROBLEMA 4: scoring de queries (nivel PRO)

👉 esto ya es diferencial SaaS

Ejemplo simple:
def score_query(result):
    if result is None:
        return 0

    if len(result) == 0:
        return 0.3

    if len(result.columns) > 1:
        return 0.8

    return 0.6
Mostrarlo:
score = score_query(result)

st.metric("🧠 Calidad de respuesta", f"{score*100:.0f}%")
