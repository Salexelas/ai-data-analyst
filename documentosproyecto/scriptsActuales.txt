import json
import re

def extract_json(text: str):

    match = re.search(r"<json>(.*?)</json>", text, re.DOTALL)

    if not match:
        raise ValueError("No JSON found")

    return match.group(1)


def parse_plan(raw: str):

    cleaned = extract_json(raw)

    try:
        return json.loads(cleaned)
    except json.JSONDecodeError:
        raise ValueError("Invalid JSON from LLM")

def create_plan(question, schema, semantic_layer=None):

    for _ in range(2):  # retry SOLO si falla LLM output

        raw = generate_raw_plan(question, schema, semantic_layer)

        try:
            plan = parse_plan(raw)
        except:
            continue  # si el LLM devuelve basura → retry

        validate_plan(plan)

        return plan

    raise ValueError("Planner failed after retries")



ALLOWED_TOOLS = {"sql", "insight"}

TOOL_SEQUENCE_RULES = {
    "sql": {
        "max": 1,
        "required": True
    },
    "insight": {
        "allowed_after": ["sql"]
    }
}


def validate_plan_structure(plan):

    if not isinstance(plan, dict):
        raise ValueError("Plan must be dict")

    steps = plan.get("steps")

    if not isinstance(steps, list) or len(steps) == 0:
        raise ValueError("Steps must be non-empty list")

    used_tools = []

    for step in steps:

        if not isinstance(step, dict):
            raise ValueError("Step must be dict")

        tool = step.get("tool")
        input_data = step.get("input")

        if tool not in ALLOWED_TOOLS:
            raise ValueError(f"Invalid tool: {tool}")

        if not isinstance(input_data, str):
            raise ValueError("Step input must be string")

        used_tools.append(tool)

    # 🔥 reglas de negocio
    if used_tools.count("sql") != 1:
        raise ValueError("Must contain exactly 1 SQL step")

    if "insight" in used_tools and used_tools[-1] != "insight":
        raise ValueError("Insight must be last step")

    return True

    return True

def execute_plan(plan, connector):

    context = {}

    # -------------------------
    # 1. Ejecutar pasos normales
    # -------------------------
    for step in plan["steps"]:

        tool_name = step["tool"]

        if tool_name not in REGISTRY:
            raise ValueError(f"Tool {tool_name} no registrada")

        entry = REGISTRY[tool_name]

        handler = entry["handler"]
        tool = entry["tool"]

        handler(step, context, connector, tool)

    # -------------------------
    # 2. Validar que hubo SQL
    # -------------------------
    if "last_result" not in context:
        raise ValueError("No hubo ejecución SQL")

    # -------------------------
    # 3. 🔥 FORZAR INSIGHT (modo test)
    # -------------------------
    if "explanation" not in context:

        insight_tool = REGISTRY["insight"]["tool"]

        try:
            result = insight_tool.run(
                "Explain the result",
                connector,
                context
            )

            context["explanation"] = result

        except Exception as e:
            context["explanation"] = f"No se pudo generar insight: {str(e)}"

    # -------------------------
    # 4. DEBUG
    # -------------------------
    print("CONTEXT FINAL:", context)

    # -------------------------
    # 5. Response
    # -------------------------
    return {
        "type": "table",
        "data": context["last_result"],
        "explanation": context.get("explanation")
    }


from agents.sql_validator import validate_sql

MAX_ROWS = 1000


def handle_sql(step, context, connector, tool):

    query = step["input"]

    schema = connector.get_schema()
    columns = schema["column_name"].tolist()

    validate_sql(query, columns)

    result = tool.run(query, connector, context)

    if result is None:
        raise ValueError("SQL devolvió None")

    if len(result) == 0:
        raise ValueError("La consulta no devolvió resultados")

    if len(result) > MAX_ROWS:
        raise ValueError("Resultado demasiado grande")

    context["last_result"] = result


def handle_insight(step, context, connector, tool):

    if "last_result" not in context:
        raise ValueError("Insight sin datos")

    result = tool.run(
        step["input"],   # 🔥 SOLO string
        connector,
        context
    )

    context["explanation"] = result

    return result

def answer_question(question, connector, semantic_layer=None):

    try:
        schema = connector.get_schema()

        plan = create_plan(question, schema, semantic_layer)

        logger.info("Plan generado", extra={
            "question": question,
            "plan": plan
        })

        return execute_plan(plan, connector)

    except ValueError as e:
        return {"type": "error", "message": str(e)}

    except Exception as e:
        return {"type": "error", "message": f"Error inesperado: {str(e)}"}




import sqlglot

ALLOWED_TABLE = "data"


def validate_sql(query: str, schema_columns: list):

    try:
        parsed = sqlglot.parse_one(query)
    except Exception:
        raise ValueError("SQL inválido")

    if parsed.key != "select":
        raise ValueError("Solo SELECT permitido")

    tables = {t.name for t in parsed.find_all(sqlglot.exp.Table)}
    if tables != {ALLOWED_TABLE}:
        raise ValueError("Tabla inválida")

    used_columns = {c.name for c in parsed.find_all(sqlglot.exp.Column)}

    invalid = used_columns - set(schema_columns)

    if invalid:
        raise ValueError(f"Columnas inválidas: {invalid}")

    return True

def call_llm(prompt: str, system_prompt: str = "You are a helpful AI") -> str:

    payload = {
        "model": "llama-3.3-70b-versatile",
        "messages": [
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": prompt}
        ],
        "temperature": 0.2
    }

    response = requests.post(API_URL, headers=headers, json=payload)

    if response.status_code != 200:
        raise Exception(response.text)

    result = response.json()

    return result["choices"][0]["message"]["content"]

def generate_semantic_suggestions(columns):

    prompt = f"""
You are a data analyst.

Given these columns:
{columns}

Generate business-friendly aliases and useful metrics.

Return ONLY valid JSON.
"""

    raw = call_llm(prompt)

    # 🔥 DEBUG ACÁ
    print("RAW LLM:", raw)

    cleaned = extract_json(raw)

    try:
        return json.loads(cleaned)
    except:
        return {"aliases": {}, "metrics": {}}




def build_semantic_layer(columns, aliases=None, metrics=None):
    return {
        "columns": columns,
        "aliases": aliases or {},
        "metrics": metrics or {}
    }


def merge_semantic_layers(base_layer, extra_layer):
    # Si uno es None → devolver el otro
    if not base_layer:
        return extra_layer

    if not extra_layer:
        return base_layer

    merged = {
        "columns": list(set(base_layer.get("columns", []) + extra_layer.get("columns", []))),
        "aliases": {},
        "metrics": {}
    }

    # -------------------------
    # MERGE ALIASES
    # -------------------------
    merged["aliases"] = base_layer.get("aliases", {}).copy()

    for col, values in extra_layer.get("aliases", {}).items():
        if col in merged["aliases"]:
            merged["aliases"][col] = list(set(merged["aliases"][col] + values))
        else:
            merged["aliases"][col] = values

    # -------------------------
    # MERGE METRICS
    # -------------------------
    merged["metrics"] = base_layer.get("metrics", {}).copy()

    # 🔥 extra pisa base si hay conflicto
    merged["metrics"].update(extra_layer.get("metrics", {}))

    return merged


import json
import os
import hashlib

def get_dataset_id(source_type, config):
    base = f"{source_type}_{str(config)}"
    return hashlib.md5(base.encode()).hexdigest()

def save_semantic_layer(dataset_id, semantic_layer):
    os.makedirs("semantic_store", exist_ok=True)
    path = f"semantic_store/{dataset_id}.json"

    with open(path, "w") as f:
        json.dump(semantic_layer, f, indent=2)

def load_semantic_layer(dataset_id):
    path = f"semantic_store/{dataset_id}.json"

    if os.path.exists(path):
        with open(path) as f:
            return json.load(f)
    return None

def list_semantic_files():
    path = "semantic_store"
    if not os.path.exists(path):
        return []
    return [f for f in os.listdir(path) if f.endswith(".json")]

def load_semantic_file(file_name):
    path = os.path.join("semantic_store", file_name)
    with open(path) as f:
        return json.load(f)

        prompt = f"""
Explain the result in simple business terms.

Question:
{question}

Data:
{preview}

Rules:
- Be concise
- No SQL
- Focus on insights
"""

        return call_llm(
            prompt,
            system_prompt="You are a senior business data analyst"
        )

def render_semantic_ui(connector):

    preview = connector.preview()
    columns = list(preview.columns)

    st.subheader("🧠 Columnas")
    st.write(columns)

    st.subheader("📊 Tipos")
    st.write(preview.dtypes)

    # -------------------------
    # MULTI SEMANTIC
    # -------------------------

    st.subheader("🧠 Usar Semantic Layers existentes")

    semantic_files = list_semantic_files()

    if semantic_files:

        col1, col2 = st.columns(2)

        with col1:
            sem1_name = st.selectbox("Semantic 1", ["None"] + semantic_files)

        with col2:
            sem2_name = st.selectbox("Semantic 2", ["None"] + semantic_files)

        if st.button("📥 Aplicar Semantic Layers"):

            sem1 = load_semantic_file(sem1_name) if sem1_name != "None" else None
            sem2 = load_semantic_file(sem2_name) if sem2_name != "None" else None

            merged = merge_semantic_layers(sem1, sem2)

            st.session_state["semantic_layer"] = merged

            st.success("✅ Semantic layers aplicadas")
            st.json(merged)

    # -------------------------
    # GENERAR
    # -------------------------

    st.subheader("✨ Generar nueva semántica")

    if st.button("Generar automáticamente"):
        raw = generate_semantic_suggestions(columns)

        st.session_state["semantic_layer"] = build_semantic_layer(
            columns,
            raw.get("aliases"),
            raw.get("metrics")
        )

        st.success("✅ Generada")

    # -------------------------
    # EDITAR
    # -------------------------

    if "semantic_layer" in st.session_state:

        sl = st.session_state["semantic_layer"]

        st.write("### 🔤 Aliases")
        st.json(sl["aliases"])

        st.write("### 📊 Métricas")
        st.json(sl["metrics"])

        st.write("### ✏️ Editar Aliases")

        for col in columns:
            current = sl["aliases"].get(col, [])
            key_name = f"alias_{col}"

            if key_name not in st.session_state:
                st.session_state[key_name] = ", ".join(current)

            value = st.text_input(col, key=key_name)
            sl["aliases"][col] = [v.strip() for v in value.split(",") if v]

        st.write("### ✏️ Editar Métricas")

        for k, v in list(sl["metrics"].items()):
            key_name = f"metric_{k}"

            if key_name not in st.session_state:
                st.session_state[key_name] = v

            new_val = st.text_input(k, key=key_name)
            sl["metrics"][k] = new_val

        # AGREGAR
        st.write("### ➕ Nueva Métrica")

        new_name = st.text_input("Nombre")
        new_expr = st.text_input("SQL")

        if st.button("Agregar métrica"):
            if new_name and new_expr:
                sl["metrics"][new_name] = new_expr
                st.success(f"✅ {new_name} agregada")

        # GUARDAR
        if st.button("💾 Guardar capa semántica"):
            save_semantic_layer(
                st.session_state["dataset_id"],
                sl
            )
            st.success("✅ Guardado")

        st.session_state["semantic_layer"] = sl

def render_source_selector():
    source_type = st.selectbox(
        "Seleccioná tipo de fuente",
        ["duckdb", "postgres", "mysql", "excel", "csv", "google_sheets", "api"]
    )

    config = {}

    if source_type == "duckdb":
        use_demo = st.checkbox("Usar tabla demo", value=True)

        uploaded_file = None
        if not use_demo:
            uploaded_file = st.file_uploader("Subí CSV/Parquet", type=["csv", "parquet"])

        if uploaded_file:
            config["file"] = uploaded_file
        elif use_demo:
            config["file"] = None

    elif source_type in ["postgres", "mysql"]:
        use_demo = st.checkbox("Usar demo", value=True)

        if use_demo:
            config["uri"] = None
        else:
            default_uri = (
                "postgresql://user:pass@localhost:5432/db"
                if source_type == "postgres"
                else "mysql+pymysql://user:pass@localhost:3306/db"
            )
            config["uri"] = st.text_input("Connection String", default_uri)

    elif source_type == "google_sheets":
        config["url"] = st.text_input("Google Sheet URL")

    elif source_type == "api":
        config["base_url"] = st.text_input("Base URL", "https://api.example.com")

    elif source_type in ["excel", "csv"]:
        uploaded_file = st.file_uploader("📂 Subí archivo", type=["csv", "xlsx"])

        if uploaded_file:
            config["file"] = uploaded_file

    if st.button("🚀 Conectar"):
        connector = create_connector(source_type, config)

        st.session_state["connector"] = connector
        dataset_id = get_dataset_id(source_type, config)
        st.session_state["dataset_id"] = dataset_id

        loaded = load_semantic_layer(dataset_id)
        if loaded:
            st.session_state["semantic_layer"] = loaded
            st.info("📂 Capa semántica cargada")

        st.success("✅ Conectado")
