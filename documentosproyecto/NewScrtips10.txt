import streamlit as st

from services.speech.loader import load_whisper
from agents.service import answer_question
from core.query_store import save_query

from ui.components.audio_input import render_audio_input
from renderers.streamlit_renderer import render_analytics


def render_chat():

    st.title("💬 Nueva conversación")

    tabs = st.tabs([
        "💬 Chat",
        "🔀 Comparación",
        "📈 Vista"
    ])

    with tabs[0]:

        default_q = st.session_state.get(
            "prefill_question",
            ""
        )

        text_prompt = st.text_input(
            "Consulta sobre dataset",
            value=default_q
        )

        audio = render_audio_input()

        if "audio_prompt" not in st.session_state:
            st.session_state["audio_prompt"] = ""

        if audio and "bytes" in audio:

            with st.spinner("🎧 Transcribiendo..."):

                transcription = load_whisper().transcribe(
                    audio["bytes"]
                )

            st.session_state["audio_prompt"] = transcription

        if st.session_state["audio_prompt"]:

            st.info(
                f"🎤 {st.session_state['audio_prompt']}"
            )

        question = (
            text_prompt
            or st.session_state["audio_prompt"]
        )

        col1, col2 = st.columns([1, 1])

        with col1:

            if st.button("Ejecutar"):

                if "connector" not in st.session_state:

                    st.error("Primero cargá un dataset")
                    return

                if not question:

                    st.warning("Ingresá una pregunta")
                    return

                st.session_state.setdefault(
                    "recent_queries",
                    []
                )

                if question not in st.session_state["recent_queries"]:

                    st.session_state["recent_queries"].insert(
                        0,
                        question
                    )

                save_query(
                    question,
                    dataset_id=st.session_state.get(
                        "dataset_id"
                    ),
                    intent=None
                )

                with st.spinner("🧠 Analizando..."):

                    response = answer_question(
                        question=question,
                        connector=st.session_state["connector"],
                        dataset_id=st.session_state.get("dataset_id"),
                        semantic_layer=st.session_state.get("semantic_layer"),
                        llm=st.session_state["llm"]
                    )

                st.session_state["last_response"] = response

        with col2:

            if st.button("Limpiar"):

                st.session_state["prefill_question"] = ""
                st.session_state["audio_prompt"] = ""

                st.session_state.pop(
                    "last_response",
                    None
                )

        st.markdown("---")

        response = st.session_state.get("last_response")

        if response:

            if response.type == "error":
                st.error(response.message)
            else:
                render_analytics(response)

        else:
            st.info("Todavía no ejecutaste ninguna consulta")

    with tabs[1]:
        st.markdown("### 🔀 Comparar datasets")
        st.info("Próximamente 🚧")

    with tabs[2]:
        st.markdown("### 📈 Opciones de vista")
        st.checkbox("Modo compacto")
        st.checkbox("Mostrar insight")
        st.checkbox("Mostrar SQL")

import streamlit as st
import os
from datetime import datetime

from services.ocr.loader import load_ocr
from services.ocr.llm_parser_service import OCRLLMParserService
from services.ocr.validator_service import OCRValidatorService
from connectors.factory import create_connector

DATA_PATH = "data"
os.makedirs(DATA_PATH, exist_ok=True)


def render_ocr_studio():

    st.title("📷 OCR Studio")

    uploaded = st.file_uploader(
        "Subir archivo",
        type=["png", "jpg", "jpeg", "pdf"]
    )

    if not uploaded:
        return

    # limpiar estado si cambia archivo
    current_file = uploaded.name
    if st.session_state.get("ocr_last_file") != current_file:
        st.session_state["ocr_last_file"] = current_file
        st.session_state["ocr_raw_text"] = ""
        st.session_state.pop("ocr_df", None)

    # =========================
    # OCR
    # =========================
    if "ocr_raw_text" not in st.session_state:
        st.session_state["ocr_raw_text"] = ""

    if st.button("Leer archivo"):

        with st.spinner("Procesando OCR..."):

            ocr = load_ocr()

            # IMPORTANTE: pasar solo bytes
            text = ocr.extract_text(
                uploaded.getvalue()
            )

            st.session_state["ocr_raw_text"] = text

    # =========================
    # RAW TEXT
    # =========================
    if st.session_state["ocr_raw_text"]:

        st.subheader("Texto detectado (editable)")

        editable_text = st.text_area(
            "Corregí si hace falta",
            value=st.session_state["ocr_raw_text"],
            height=320
        )

        context = st.text_input(
            "Contexto",
            placeholder="Ej: factura ferretería / estadísticas NBA"
        )

        columns = st.text_input(
            "Columnas",
            placeholder="Ej: producto,cantidad,precio_unitario"
        )

        examples = st.text_area(
            "Ejemplos (opcional)",
            placeholder="Ej:\nPintura,3,15500\nCeramica,2,12000"
        )

        # =========================
        # GENERATE DATAFRAME
        # =========================
        if st.button("Generar dataset"):

            cols = [
                c.strip()
                for c in columns.split(",")
                if c.strip()
            ]

            if not cols:
                st.warning("Ingresá al menos una columna")
                return

            parser = OCRLLMParserService()

            with st.spinner("Estructurando dataset..."):

                df = parser.parse_to_dataframe(
                    raw_text=editable_text,
                    columns=cols,
                    examples=examples,
                    context=context,
                    llm=st.session_state["llm"]
                )

            st.session_state["ocr_df"] = df

    # =========================
    # PREVIEW
    # =========================
    if "ocr_df" in st.session_state:

        st.subheader("Preview dataset")

        edited_df = st.data_editor(
            st.session_state["ocr_df"],
            use_container_width=True,
            num_rows="dynamic"
        )

        validator = OCRValidatorService()

        errors = validator.validate_dataframe(
            edited_df
        )

        if errors:

            for err in errors:
                st.error(err)

        else:

            st.success("Dataset válido ✔")

            if st.button("Guardar dataset"):

                file_name = datetime.now().strftime(
                    "ocr_%Y%m%d_%H%M%S"
                ) + ".csv"

                save_path = os.path.join(
                    DATA_PATH,
                    file_name
                )

                edited_df.to_csv(
                    save_path,
                    index=False
                )

                connector = create_connector(
                    "csv",
                    {"file_path": save_path}
                )

                st.session_state["connector"] = connector
                st.session_state["file_path"] = save_path
                st.session_state["dataset_id"] = file_name

                st.success(
                    f"Dataset guardado: {file_name}"
                )

                st.rerun()



llm_parser_service.py
import pandas as pd
import json
import re


class OCRLLMParserService:

    def clean_json(self, response):

        response = response.strip()
        response = response.replace("```json", "")
        response = response.replace("```", "")

        match = re.search(
            r"\[.*\]",
            response,
            re.DOTALL
        )

        if match:
            return match.group(0)

        return response

    def parse_to_dataframe(
        self,
        raw_text,
        columns,
        examples,
        context,
        llm
    ):

        if not columns:
            return pd.DataFrame({
                "error": ["No columns"]
            })

        prompt = f"""
Extraé dataset estructurado.

CONTEXTO:
{context}

TEXTO OCR:
{raw_text}

COLUMNAS:
{", ".join(columns)}

EJEMPLOS:
{examples}

REGLAS:
- devolver SOLO JSON
- no inventar
- null si falta
"""

        response = llm.generate(
            prompt=prompt,
            system_prompt="OCR parser"
        )

        cleaned = self.clean_json(response)

        try:
            return pd.DataFrame(json.loads(cleaned))

        except Exception:
            return pd.DataFrame({
                "raw_output": [response]
            })


paddleocr_service.py
import tempfile
from paddleocr import PaddleOCR


class OCRService:

    def __init__(self):

        self.ocr = PaddleOCR(
            use_angle_cls=True,
            lang="en"
        )

    def extract_text(self, file_bytes):

        with tempfile.NamedTemporaryFile(
            delete=False,
            suffix=".png"
        ) as f:

            f.write(file_bytes)
            temp_path = f.name

        result = self.ocr.ocr(temp_path)

        lines = []

        for page in result:
            for item in page:

                text = item[1][0]
                score = item[1][1]

                if score > 0.4:
                    lines.append(text)

        return "\n".join(lines)
