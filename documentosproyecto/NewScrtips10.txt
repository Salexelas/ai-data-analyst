import streamlit as st

from agents.service import answer_question
from core.query_store import save_query

def render_chat():

    st.title("💬 Nueva conversación")

    tabs = st.tabs([
        "💬 Chat",
        "🔀 Comparación",
        "📈 Vista"
    ])

    # =========================
    # TAB CHAT
    # =========================
    with tabs[0]:

        default_q = st.session_state.get(
            "prefill_question",
            ""
        )

        # =========================
        # 🔹 INPUT TEXTO
        # =========================
        text_prompt = st.text_input(
            "Escribí tu pregunta...",
            value=default_q
        )

        # =========================
        # 🔹 AUDIO
        # =========================
        audio = render_audio_input()

        # =========================
        # INIT
        # =========================
        if "audio_prompt" not in st.session_state:

            st.session_state["audio_prompt"] = ""

        # =========================
        # TRANSCRIBE
        # =========================
        if audio and "bytes" in audio:

            with st.spinner("🎧 Transcribiendo..."):

                transcription = (
                    st.session_state["whisper"]
                    .transcribe(audio["bytes"])
                )

            st.session_state[
                "audio_prompt"
            ] = transcription

        # =========================
        # SHOW AUDIO
        # =========================
        if st.session_state["audio_prompt"]:

            st.info(
                f"🎤 {st.session_state['audio_prompt']}"
            )

        # =========================
        # FINAL QUESTION
        # =========================
        question = (
            text_prompt
            or st.session_state["audio_prompt"]
        )

        # =========================
        # BUTTONS
        # =========================
        col1, col2 = st.columns([1, 1])

        # =========================
        # EXECUTE
        # =========================
        with col1:

            if st.button("Ejecutar"):

                # -------------------------
                # DATASET VALIDATION
                # -------------------------
                if "connector" not in st.session_state:

                    st.error(
                        "Primero cargá un dataset"
                    )

                    return

                # -------------------------
                # QUESTION VALIDATION
                # -------------------------
                if not question:

                    st.warning(
                        "Ingresá una pregunta"
                    )

                    return

                # -------------------------
                # RECENTS
                # -------------------------
                st.session_state.setdefault(
                    "recent_queries",
                    []
                )

                if (
                    question
                    not in st.session_state[
                        "recent_queries"
                    ]
                ):

                    st.session_state[
                        "recent_queries"
                    ].insert(
                        0,
                        question
                    )

                # -------------------------
                # SAVE QUERY
                # -------------------------
                save_query(
                    question,
                    dataset_id=st.session_state.get(
                        "dataset_id"
                    ),
                    intent=None
                )

                # -------------------------
                # PIPELINE
                # -------------------------
                with st.spinner(
                    "🧠 Analizando..."
                ):

                    response = answer_question(

                        question=question,

                        connector=st.session_state[
                            "connector"
                        ],

                        dataset_id=st.session_state.get(
                            "dataset_id"
                        ),

                        semantic_layer=st.session_state.get(
                            "semantic_layer"
                        ),

                        llm=st.session_state[
                            "llm"
                        ]
                    )

                st.session_state[
                    "last_response"
                ] = response

        # =========================
        # CLEAR
        # =========================
        with col2:

            if st.button("Limpiar"):

                st.session_state[
                    "prefill_question"
                ] = ""

                st.session_state[
                    "audio_prompt"
                ] = ""

                st.session_state.pop(
                    "last_response",
                    None
                )

        st.markdown("---")

        # =========================
        # RESULT
        # =========================
        response = st.session_state.get(
            "last_response"
        )

        if response:

            # -------------------------
            # ERROR
            # -------------------------
            if response.type == "error":

                st.error(
                    response.message
                )

            # -------------------------
            # ANALYTICS RESPONSE
            # -------------------------
            else:

                render_analytics(response)

        else:

            st.info(
                "Todavía no ejecutaste ninguna consulta"
            )

    # =========================
    # TAB COMPARACIÓN
    # =========================
    with tabs[1]:

        st.markdown(
            "### 🔀 Comparar datasets"
        )

        st.info(
            "Próximamente 🚧"
        )

    # =========================
    # TAB VISTA
    # =========================
    with tabs[2]:

        st.markdown(
            "### 📈 Opciones de vista"
        )

        st.checkbox(
            "Modo compacto"
        )

        st.checkbox(
            "Mostrar insight"
        )

        st.checkbox(
            "Mostrar SQL"
        )

import streamlit as st
import os
from connectors.factory import create_connector
from semantic.utils import get_dataset_id, load_semantic_layer

# 📁 carpeta persistente
DATA_PATH = "data"
os.makedirs(DATA_PATH, exist_ok=True)


def render_data():

    st.title("📥 Gestión de Datos")

    tabs = st.tabs([
        "📂 Subir archivo",
        "🔗 Conectar fuente",
        "📊 Preview"
    ])

    # -------------------------
    # 📂 UPLOAD (CSV / Excel)
    # -------------------------
    with tabs[0]:

        file = st.file_uploader("Subir CSV / Excel", type=["csv", "xlsx"])

        if file and st.button("Cargar archivo"):

            file_path = os.path.join(DATA_PATH, file.name)

            # 💾 guardar archivo en disco
            with open(file_path, "wb") as f:
                f.write(file.getbuffer())

            file_type = "csv" if file.name.endswith(".csv") else "excel"

            # ⚠️ ahora usamos file_path (persistente)
            connector = create_connector(file_type, {"file_path": file_path})

            st.session_state["connector"] = connector
            st.session_state["file_path"] = file_path

            dataset_id = get_dataset_id(file_type, {"file": file.name})
            st.session_state["dataset_id"] = dataset_id

            loaded = load_semantic_layer(dataset_id)
            if loaded:
                st.session_state["semantic_layer"] = loaded

            st.success(f"✅ Dataset guardado en {file_path}")

    # -------------------------
    # 🔗 CONNECT (DB / API)
    # -------------------------
    with tabs[1]:

        source_type = st.selectbox(
            "Tipo de fuente",
            ["postgres", "mysql", "api", "google_sheets"]
        )

        config = {}

        if source_type in ["postgres", "mysql"]:
            config["uri"] = st.text_input("Connection string")

        elif source_type == "api":
            config["base_url"] = st.text_input("Base URL")

        elif source_type == "google_sheets":
            config["url"] = st.text_input("Google Sheet URL")

        if st.button("Conectar"):

            connector = create_connector(source_type, config)

            st.session_state["connector"] = connector

            dataset_id = get_dataset_id(source_type, config)
            st.session_state["dataset_id"] = dataset_id

            loaded = load_semantic_layer(dataset_id)
            if loaded:
                st.session_state["semantic_layer"] = loaded

            st.success("✅ Conectado")

    # -------------------------
    # 📊 PREVIEW
    # -------------------------
    with tabs[2]:

        if "connector" in st.session_state:

            st.info("Vista previa del dataset")

            preview = st.session_state["connector"].preview()
            st.dataframe(preview)

        else:
            st.warning("Primero cargá una fuente de datos")

import streamlit as st
import pandas as pd
import os
import json

from connectors.factory import create_connector

# ==================================================
# PATHS
# ==================================================
DATA_PATH = "data"
SCHEMA_PATH = "schemas"

os.makedirs(DATA_PATH, exist_ok=True)
os.makedirs(SCHEMA_PATH, exist_ok=True)

# ==================================================
# COLUMN TYPES
# ==================================================
TYPE_OPTIONS = [
    "auto",
    "text",
    "number",
    "datetime",
    "boolean",
    "category",
    "id"
]

# ==================================================
# LOAD SCHEMA
# ==================================================
def load_schema(dataset_name):

    schema_file = os.path.join(
        SCHEMA_PATH,
        f"{dataset_name}.schema.json"
    )

    if not os.path.exists(schema_file):
        return {}

    try:

        with open(schema_file, "r", encoding="utf-8") as f:
            return json.load(f)

    except:
        return {}

# ==================================================
# SAVE SCHEMA
# ==================================================
def save_schema(dataset_name, schema):

    schema_file = os.path.join(
        SCHEMA_PATH,
        f"{dataset_name}.schema.json"
    )

    with open(schema_file, "w", encoding="utf-8") as f:

        json.dump(
            schema,
            f,
            indent=4,
            ensure_ascii=False
        )

# ==================================================
# DETECT DEFAULT TYPE
# ==================================================
def detect_default_type(series):

    dtype = str(series.dtype)

    if "datetime" in dtype:
        return "datetime"

    if "int" in dtype or "float" in dtype:
        return "number"

    if "bool" in dtype:
        return "boolean"

    return "text"

# ==================================================
# APPLY COLUMN TYPES
# ==================================================
def apply_column_types(df, schema):

    for col, dtype in schema.items():

        if col not in df.columns:
            continue

        try:

            # -------------------------
            # DATETIME
            # -------------------------
            if dtype == "datetime":

                df[col] = pd.to_datetime(
                    df[col],
                    dayfirst=True,
                    errors="coerce"
                )

            # -------------------------
            # NUMBER
            # -------------------------
            elif dtype == "number":

                df[col] = pd.to_numeric(
                    df[col],
                    errors="coerce"
                )

            # -------------------------
            # BOOLEAN
            # -------------------------
            elif dtype == "boolean":

                df[col] = df[col].astype("bool")

            # -------------------------
            # CATEGORY
            # -------------------------
            elif dtype == "category":

                df[col] = df[col].astype("category")

            # -------------------------
            # TEXT
            # -------------------------
            elif dtype == "text":

                df[col] = df[col].astype("string")

        except Exception as e:

            st.warning(f"Error casteando {col}: {e}")

    return df

# MAIN
# ==================================================
def render_dataset_editor():

    st.title("✏️ Editor de Dataset")

    # ==================================================
    # SESSION
    # ==================================================
    file_path = st.session_state.get("file_path")

    if not file_path:

        st.warning("Primero cargá un dataset")
        return

    if not os.path.exists(file_path):

        st.error("El archivo no existe")
        return

    file_name = os.path.basename(file_path)

    # ==================================================
    # FILE TYPE
    # ==================================================
    if file_name.endswith(".csv"):

        file_type = "csv"

    elif file_name.endswith(".xlsx"):

        file_type = "excel"

    else:

        st.error("Formato no soportado")
        return

    # ==================================================
    # LOAD DATAFRAME
    # ==================================================
    try:

        connector = create_connector(
            file_type,
            {"file_path": file_path}
        )

        df = connector.preview(100000)

    except Exception as e:

        st.error(f"Error cargando dataset: {e}")
        return

    # ==================================================
    # LOAD EXISTING SCHEMA
    # ==================================================
    schema = load_schema(file_name)

    # ==================================================
    # TABS
    # ==================================================
    tabs = st.tabs([
        "📝 Columnas",
        "🧠 Tipos",
        "📊 Datos"
    ])

    # ==================================================
    # 📝 COLUMNAS
    # ==================================================
    with tabs[0]:

        st.subheader("📝 Renombrar columnas")

        rename_map = {}

        cols = st.columns(2)

        for i, col in enumerate(df.columns):

            with cols[i % 2]:

                new_name = st.text_input(
                    f"{col}",
                    value=col,
                    key=f"rename_{col}"
                )

                rename_map[col] = new_name

    # ==================================================
    # 🧠 TIPOS
    # ==================================================
    with tabs[1]:

        st.subheader("🧠 Tipos de columnas")

        column_types = {}

        cols = st.columns(2)

        for i, col in enumerate(df.columns):

            with cols[i % 2]:

                saved_type = schema.get(col)

                default_type = (
                    saved_type
                    if saved_type
                    else detect_default_type(df[col])
                )

                column_types[col] = st.selectbox(
                    f"Tipo → {col}",
                    TYPE_OPTIONS,
                    index=TYPE_OPTIONS.index(default_type),
                    key=f"type_{col}"
                )

    # ==================================================
    # APPLY TYPES
    # ==================================================
    df = apply_column_types(
        df,
        column_types
    )

    # ==================================================
    # 📊 DATOS
    # ==================================================
    with tabs[2]:

        st.subheader("📊 Editor")

        edited_df = st.data_editor(
            df,
            use_container_width=True,
            num_rows="dynamic",
            height=600
        )

    # ==================================================
    # APPLY RENAME
    # ==================================================
    edited_df.columns = [
        rename_map[c]
        for c in edited_df.columns
    ]

    # ==================================================
    # SAVE
    # ==================================================
    st.divider()

    if st.button("💾 Guardar cambios"):

        try:

            # ==================================================
            # SAVE DATASET
            # ==================================================
            if file_type == "csv":

                edited_df.to_csv(
                    file_path,
                    index=False
                )

            else:

                edited_df.to_excel(
                    file_path,
                    index=False
                )

            # ==================================================
            # UPDATE SCHEMA
            # ==================================================
            updated_schema = {}

            for old_col, new_col in rename_map.items():

                updated_schema[new_col] = column_types[old_col]

            save_schema(
                file_name,
                updated_schema
            )

            # ==================================================
            # RELOAD CONNECTOR
            # ==================================================
            connector = create_connector(
                file_type,
                {"file_path": file_path}
            )

            st.session_state["connector"] = connector

            # invalidate semantic layer
            st.session_state["semantic_layer"] = None

            st.success("✅ Dataset actualizado")

            st.rerun()

        except Exception as e:

            st.error(f"Error guardando dataset: {e}")

import streamlit as st
import pandas as pd


def render_dataset_info():

    st.title("📊 Dataset Analytics")

    if "connector" not in st.session_state:

        st.warning(
            "No hay dataset cargado"
        )

        return

    connector = st.session_state["connector"]

    # 🔥 dataframe REAL
    df = connector.preview(limit=100000)

    st.markdown("---")

    # =========================
    # GENERAL
    # =========================

    st.subheader(
        "📌 Información General"
    )

    col1, col2, col3, col4 = st.columns(4)

    col1.metric(
        "Rows",
        len(df)
    )

    col2.metric(
        "Columns",
        len(df.columns)
    )

    col3.metric(
        "Missing",
        int(df.isna().sum().sum())
    )

    memory_mb = round(
        df.memory_usage(deep=True).sum()
        / 1024 / 1024,
        2
    )

    col4.metric(
        "Memory MB",
        memory_mb
    )

    st.markdown("---")

    # =========================
    # SCHEMA
    # =========================

    st.subheader(
        "🧬 Schema"
    )

    schema_df = pd.DataFrame({

        "column": df.columns,

        "dtype": df.dtypes.astype(str),

        "nulls": df.isna().sum().values,

        "unique": df.nunique().values
    })

    st.dataframe(
        schema_df,
        use_container_width=True
    )

    st.markdown("---")

    # =========================
    # NUMERIC
    # =========================

    numeric_df = df.select_dtypes(
        include="number"
    )

    if not numeric_df.empty:

        st.subheader(
            "📈 Numeric Analytics"
        )

        stats = pd.DataFrame({

            "mean": numeric_df.mean(),

            "median": numeric_df.median(),

            "std": numeric_df.std(),

            "min": numeric_df.min(),

            "max": numeric_df.max(),

            "missing": numeric_df.isna().sum()
        })

        st.dataframe(
            stats.round(2),
            use_container_width=True
        )

        st.bar_chart(
            numeric_df.mean()
        )

    st.markdown("---")

    # =========================
    # CATEGORICAL
    # =========================

    cat_df = df.select_dtypes(
        exclude="number"
    )

    if not cat_df.empty:

        st.subheader(
            "🏷 Variables categóricas"
        )

        for col in cat_df.columns[:5]:

            st.markdown(f"### {col}")

            vc = (
                cat_df[col]
                .astype(str)
                .value_counts()
                .head(10)
            )

            st.bar_chart(vc)

    st.markdown("---")

    # =========================
    # CORRELATION
    # =========================

    if len(numeric_df.columns) > 1:

        st.subheader(
            "🔥 Correlaciones"
        )

        corr = numeric_df.corr(
            numeric_only=True
        )

        st.dataframe(
            corr.round(2),
            use_container_width=True
        )

    st.markdown("---")

    # =========================
    # RAW
    # =========================

    st.subheader("📄 Raw Data")

    st.dataframe(
        df.head(100),
        use_container_width=True
    )

import streamlit as st
import os

from connectors.factory import create_connector

DATA_PATH = "data"


# =========================
# HELPERS
# =========================

def detect_file_type(file):

    if file.endswith(".csv"):
        return "csv"

    elif file.endswith(".xlsx"):
        return "excel"

    return None


def load_dataset(file, file_path):

    file_type = detect_file_type(file)

    if not file_type:
        raise ValueError(
            "Formato no soportado"
        )

    connector = create_connector(
        file_type,
        {"file_path": file_path}
    )

    st.session_state["connector"] = connector

    st.session_state["file_path"] = file_path

    st.session_state["dataset_id"] = file

    return connector


# =========================
# PAGE
# =========================

def render_datasets():

    st.title("📚 Gestión de Datasets")

    # -------------------------
    # VALIDATE FOLDER
    # -------------------------
    if not os.path.exists(DATA_PATH):

        st.info(
            "No hay datasets todavía"
        )

        return

    # -------------------------
    # FILES
    # -------------------------
    files = sorted(
        os.listdir(DATA_PATH),
        reverse=True
    )

    if not files:

        st.info(
            "No hay datasets guardados"
        )

        return

    st.markdown("---")

    # -------------------------
    # LOOP
    # -------------------------
    for file in files:

        file_path = os.path.join(
            DATA_PATH,
            file
        )

        # 🔥 5 columnas ahora
        col1, col2, col3, col4, col5 = st.columns(
            [4, 1, 1, 1, 1]
        )

        # ==================================================
        # FILE NAME
        # ==================================================
        with col1:

            st.write(f"📄 {file}")

            current_file = st.session_state.get(
                "file_path"
            )

            if current_file == file_path:

                st.caption(

        # ==================================================

                    st.error(
                        f"Error eliminando: {e}"
                    )

        # ==================================================

                st.dataframe(
                    preview,
                    use_container_width=True
                )

            except Exception as e:

                st.error(
                    f"Error preview: {e}"

import streamlit as st
from semantic.semantic_layer import build_semantic_layer
from semantic.utils import save_semantic_layer

def render_semantic():

    st.title("🧠 Capa Semántica")

    if "connector" not in st.session_state:
        st.warning("Primero cargá un dataset")
        return

    connector = st.session_state["connector"]
    columns = list(connector.preview().columns)

    # -------------------------
    # INIT SI NO EXISTE
    # -------------------------
    if "semantic_layer" not in st.session_state or st.session_state["semantic_layer"] is None:

        st.session_state["semantic_layer"] = build_semantic_layer(columns)

    sl = st.session_state["semantic_layer"]

    tabs = st.tabs([
        "🔤 Aliases",
        "📊 Métricas",
        "🧪 Test"
    ])

    # -------------------------
    # 🔤 ALIASES
    # -------------------------
    with tabs[0]:

        st.subheader("Aliases por columna")

        for col in columns:

            current = sl["aliases"].get(col, [])

            key = f"alias_{col}"

            if key not in st.session_state:
                st.session_state[key] = ", ".join(current)

            value = st.text_input(f"{col}", key=key)

            sl["aliases"][col] = [v.strip() for v in value.split(",") if v]

    # -------------------------
    # 📊 METRICS
    # -------------------------
    with tabs[1]:

        st.subheader("Métricas")

        for k, v in list(sl["metrics"].items()):

            key = f"metric_{k}"

            if key not in st.session_state:
                st.session_state[key] = v

            new_val = st.text_input(k, key=key)
            sl["metrics"][k] = new_val

        st.markdown("### ➕ Nueva métrica")

        new_name = st.text_input("Nombre métrica")
        new_expr = st.text_input("SQL")

        if st.button("Agregar métrica"):
            if new_name and new_expr:
                sl["metrics"][new_name] = new_expr
                st.success(f"✅ {new_name} agregada")

    # -------------------------
    # 🧪 TEST (simple)
    # -------------------------
    with tabs[2]:

        st.subheader("Test rápido")

        st.json(sl)

    # -------------------------
    # SAVE
    # -------------------------
    if st.button("💾 Guardar capa semántica"):

        save_semantic_layer(
            st.session_state["dataset_id"],
            sl
        )

        st.success("✅ Guardado")

    # persistencia final
    st.session_state["semantic_layer"] = sl
