from core.llm import call_llm


def clean_sql(output: str) -> str:
    """
    Limpia la respuesta del modelo para quedarse solo con SQL.
    """
    if "```" in output:
        return output.split("```")[1].strip()

    if "SQL:" in output:
        return output.split("SQL:")[-1].strip()

    return output.strip()


def generate_sql(question, schema):
    prompt = f"""
You are a senior SQL expert.

ONLY return a valid SQL query.
Do not explain anything.
Do not add comments.
Do not add markdown.

Schema:
{schema}

Question:
{question}

SQL:
"""

    raw_output = call_llm(prompt)
    sql = clean_sql(raw_output)

    return sql

import duckdb
import pandas as pd
from connectors.base import BaseConnector

class DuckDBConnector(BaseConnector):
    def __init__(self, path=None):
        self.con = duckdb.connect()
        self.table_name = "data"

        if path:  # CSV/Parquet subido
            self.con.execute(f"""
                CREATE OR REPLACE TABLE {self.table_name} AS
                SELECT * FROM read_csv_auto('{path}')
            """)
        else:  # Tabla demo
            self.con.execute(f"""
                CREATE OR REPLACE TABLE {self.table_name} AS
                SELECT * FROM (VALUES
                    ('Producto A', 10, 100),
                    ('Producto B', 5, 200),
                    ('Producto C', 8, 150)
                ) AS t(producto, cantidad, precio)
            """)

    def execute(self, query):
        return self.con.execute(query).df()

    def get_schema(self):
        return self.con.execute(f"DESCRIBE {self.table_name}").df()

    def preview(self, limit=10):
        return self.con.execute(f"SELECT * FROM {self.table_name} LIMIT {limit}").df()

from connectors.duckdb import DuckDBConnector
from connectors.postgres import PostgresConnector
from connectors.mysql import MySQLConnector
from connectors.api import APIConnector
from connectors.excel import ExcelConnector
from connectors.csv import CSVConnector
from connectors.googleSheets import GoogleSheetsConnector


def create_connector(source_type, config):

    if source_type == "duckdb":
        return DuckDBConnector(config.get("file"))  # <- usar get para aceptar None

    elif source_type == "postgres":
        return PostgresConnector(config["uri"])

    elif source_type == "mysql":
        return MySQLConnector(config["uri"])

    elif source_type == "excel":
        return ExcelConnector(config["file"])

    elif source_type == "csv":
        return CSVConnector(config["file"])

    elif source_type == "api":
        return APIConnector(config["base_url"])
    
    elif source_type == "google_sheets":
        return GoogleSheetsConnector(config["url"])

    else:
        raise ValueError("Unsupported data source")

class GoogleSheetsConnector(BaseConnector):

        self.df = pd.read_csv(csv_url)
        self.conn = duckdb.connect()
        self.conn.register("data", self.df)

    def execute(self, query):
        return self.conn.execute(query).df()

    def get_schema(self):
        return self.conn.execute("DESCRIBE data").df()

    def preview(self, limit=10):
        return self.df.head(limit)
    
    def convert_to_csv_url(url: str) -> str:
        if "docs.google.com/spreadsheets" not in url:
            raise ValueError("No es un link válido de Google Sheets")

        try:
            sheet_id = url.split("/d/")[1].split("/")[0]
            return f"https://docs.google.com/spreadsheets/d/{sheet_id}/gviz/tq?tqx=out:csv"
        except:
            raise ValueError("No se pudo parsear el link")
        
def convert_to_csv_url(url: str) -> str:
    sheet_id = url.split("/d/")[1].split("/")[0]

    if "gid=" in url:
        gid = url.split("gid=")[1]
        return f"https://docs.google.com/spreadsheets/d/{sheet_id}/export?format=csv&gid={gid}"

    return f"https://docs.google.com/spreadsheets/d/{sheet_id}/export?format=csv"

import pandas as pd
import sqlalchemy as sa
from connectors.base import BaseConnector

class MySQLConnector(BaseConnector):
    def __init__(self, uri=None):
        if uri:  # conexión real
            self.engine = sa.create_engine(uri)
            self.demo = False
        else:  # demo
            self.demo = True
            self.df = pd.DataFrame({
                "producto": ["Producto A", "Producto B", "Producto C"],
                "cantidad": [10, 5, 8],
                "precio": [100, 200, 150]
            })

    def execute(self, query):
        if self.demo:
            return self.df  # simple: devuelve todo
        else:
            with self.engine.connect() as conn:
                result = conn.execute(sa.text(query))
                return pd.DataFrame(result.fetchall(), columns=result.keys())

    def get_schema(self):
        if self.demo:
            return pd.DataFrame({
                "column_name": self.df.columns,
                "data_type": [str(dt) for dt in self.df.dtypes]
            })
        else:
            query = """
            SELECT table_name, column_name, data_type
            FROM information_schema.columns
            """
            return self.execute(query)
    
    def preview(self, limit=10):
        if self.demo:
            return self.df.head(limit)
        else:
            query = f"SELECT * FROM information_schema.tables LIMIT 1"
            return self.execute(query)



postgres.py
import pandas as pd
import sqlalchemy as sa
from connectors.base import BaseConnector

class PostgresConnector(BaseConnector):
    def __init__(self, uri):
        self.engine = sa.create_engine(uri)

    def execute(self, query):
        with self.engine.connect() as conn:
            result = conn.execute(sa.text(query))
            return pd.DataFrame(result.fetchall(), columns=result.keys())

    def get_schema(self):
        query = """
        SELECT table_name, column_name, data_type
        FROM information_schema.columns
        WHERE table_schema = 'public'
        ORDER BY table_name
        """
        return self.execute(query)
    
    def preview(self, limit=10):
        tables = self.get_tables()

        if not tables:
            return pd.DataFrame({"error": ["No hay tablas en schema public"]})

        table = tables[0]

        query = f"SELECT * FROM {table} LIMIT {limit}"
        return self.execute(query)
    
    def get_tables(self):
        query = """
        SELECT table_name
        FROM information_schema.tables
        WHERE table_schema = 'public'
        """
        df = self.execute(query)
        return df["table_name"].tolist()




llm.py
import os
import requests
from dotenv import load_dotenv

load_dotenv()

API_KEY = os.getenv("HUGGINGFACE_API_KEY")

API_URL = "https://api-inference.huggingface.co/models/mistralai/Mistral-7B-Instruct-v0.2"

headers = {
    "Authorization": f"Bearer {API_KEY}"
}


def call_llm(prompt: str) -> str:
    """
    Llama al modelo de Hugging Face y devuelve texto generado.
    """

    if not API_KEY:
        raise ValueError("Falta HUGGINGFACE_API_KEY en .env")

    payload = {
        "inputs": prompt,
        "parameters": {
            "max_new_tokens": 200,
            "temperature": 0.2,
            "return_full_text": False  # 🔥 clave para evitar que repita el prompt
        }
    }

    response = requests.post(API_URL, headers=headers, json=payload)
    result = response.json()

    # Debug opcional
    # print("HF response:", result)

    try:
        return result[0]["generated_text"].strip()
    except Exception:
        return str(result)

import streamlit as st
from connectors.factory import create_connector

st.set_page_config(page_title="AI Data Analyst", layout="wide")

st.title("🔥 AI Data Analyst")
st.write("Conectá tu fuente de datos en segundos")

# -------------------------
# SELECTOR DE SOURCE
# -------------------------

source_type = st.selectbox(
    "Seleccioná tipo de fuente",
    ["duckdb", "postgres", "mysql", "excel", "csv","google_sheets" ,"api"]
)

config = {}

# -------------------------
# INPUTS DINÁMICOS
# -------------------------

if source_type == "duckdb":
    use_demo = st.checkbox("Usar tabla demo (sin subir archivo)", value=True)

    uploaded_file = None
    if not use_demo:
        uploaded_file = st.file_uploader(
            "Subí CSV/Parquet",
            type=["csv", "parquet"]
        )

    # Guardar en config
    if uploaded_file:
        config["file"] = uploaded_file
    elif use_demo:
        config["file"] = None  # DuckDBConnector detecta None y carga demo

elif source_type in ["postgres", "mysql"]:
    use_demo = st.checkbox("Usar tabla demo (sin conexión real)", value=True)

    if use_demo:
        config["uri"] = None  # el conector detecta None y carga demo
    else:
        default_uri = (
            "postgresql://user:pass@localhost:5432/db"
            if source_type == "postgres"
            else "mysql+pymysql://user:pass@localhost:3306/db"
        )
        config["uri"] = st.text_input("🔗 Pegá tu Connection String", default_uri)

elif source_type == "google_sheets":
    config["url"] = st.text_input("Pegá tu Google Sheet")

elif source_type == "api":
    config["base_url"] = st.text_input("Base URL", "https://api.example.com")

# 🔥 ACÁ ESTÁ LO IMPORTANTE (drag & drop)
elif source_type in ["excel", "csv"]:

    uploaded_file = st.file_uploader(
        "📂 Arrastrá tu archivo (CSV o Excel)",
        type=["csv", "xlsx"]
    )

    if uploaded_file is not None:
        config["file"] = uploaded_file


# -------------------------
# CONNECT BUTTON
# -------------------------

if st.button("🚀 Conectar"):

    try:
        # Validación para archivos
        if source_type in ["excel", "csv"] and "file" not in config:
            st.warning("⚠️ Subí un archivo primero")
            st.stop()

        connector = create_connector(source_type, config)

        st.success("✅ Conexión exitosa")

        # -------------------------
        # SCHEMA
        # -------------------------
        st.subheader("📚 Schema")
        schema = connector.get_schema()
        st.dataframe(schema)

        # -------------------------
        # PREVIEW
        # -------------------------
        st.subheader("👀 Preview")
        preview = connector.preview()
        st.dataframe(preview)

        # Columnas detectadas (pro level)
        st.subheader("🧠 Columnas detectadas")
        st.write(list(preview.columns))

        st.subheader("📊 Tipos de datos")
        st.write(preview.dtypes)

        # Guardar en session
        st.session_state["connector"] = connector

    except Exception as e:
        st.error(f"❌ Error de conexión: {str(e)}")


# -------------------------
# QUERY SECTION
# -------------------------

if "connector" in st.session_state:

    st.subheader("💬 Hacer pregunta")

    question = st.text_input("Pregunta")

    if st.button("Ejecutar"):

        try:
            from agents.router import route_question

            result = route_question(
                question,
                st.session_state["connector"]
            )

            st.subheader("📊 Resultado")
            st.dataframe(result)

        except Exception as e:
            st.error(f"❌ Error en la query: {str(e)}")
