import unicodedata
import re

def normalize_query_for_intent(text: str):
    text = normalize(text)  # ya tenés esto

    replacements = {
        "más": "mas",
        "top": "top",
        "mayores": "top",
        "mejores": "top",
        "peores": "bottom",
        "menos": "bottom"
    }

    tokens = text.split()
    tokens = [replacements.get(t, t) for t in tokens]

    return " ".join(tokens)


def fuzzy_match_score(q, term):
    q_tokens = set(q.split())
    t_tokens = set(term.split())

    if not t_tokens:
        return 0

    overlap = len(q_tokens & t_tokens)
    return overlap / len(t_tokens)

def normalize(text):
    text = text.lower()
    text = unicodedata.normalize("NFKD", text)
    text = "".join(c for c in text if not unicodedata.combining(c))
    text = re.sub(r"[^a-z0-9\s]", "", text)
    return text


def is_match(q, term):
    return term in q or q in term


def token_overlap(a, b):
    a_tokens = set(a.split())
    b_tokens = set(b.split())
    return len(a_tokens & b_tokens) > 0


def is_valid_question_basic(question: str):
    q = question.strip().lower()

    if len(q) < 3:
        return False

    if not any(c.isalpha() for c in q):
        return False

    return True


def get_search_terms(semantic_layer):

    terms = list(
        semantic_layer.get("metrics", {}).keys()
    )

    for aliases in semantic_layer.get(
        "aliases",
        {},
    ).values():

        if isinstance(aliases, list):
            terms.extend(aliases)
        else:
            terms.append(aliases)

    terms.extend(
        semantic_layer.get("columns", [])
    )

    return [
        normalize(term)
        for term in terms
    ]


def match_score(question, term):

    question_tokens = set(question.split())
    term_tokens = set(term.split())

    if not term_tokens:
        return 0

    coverage = (
        len(question_tokens & term_tokens)
        / len(term_tokens)
    )

    substring = int(term in question)

    return coverage + 0.3 * substring


def is_question_related_fast(
    question,
    semantic_layer,
):

    if not semantic_layer:
        return True

    question = normalize(question)

    return any(
        match_score(question, term) >= 0.5
        for term in get_search_terms(
            semantic_layer
        )
    )


def detect_language_fast(text):
    text = text.lower()

    scores = {
        "es": 0,
        "en": 0,
        "pt": 0,
        "de": 0,
        "fr": 0,
        "it": 0
    }

    lang_keywords = {
        "es": ["el", "la", "los", "las", "de", "que", "en", "y", "para", "con", "por", "estadisticas", "equipo"],
        "en": ["the", "is", "are", "of", "and", "in", "for", "with", "stats", "team"],
        "pt": ["o", "a", "os", "as", "de", "que", "em", "para", "com", "estatisticas", "equipe"],
        "de": ["der", "die", "das", "und", "ist", "in", "für", "mit", "statistik", "mannschaft"],
        "fr": ["le", "la", "les", "de", "et", "est", "dans", "pour", "avec", "statistiques", "equipe"],
        "it": ["il", "lo", "la", "gli", "le", "di", "e", "è", "per", "con", "statistiche", "squadra"]
    }

    tokens = text.split()

    for lang, keywords in lang_keywords.items():
        for token in tokens:
            if token in keywords:
                scores[lang] += 1

    best_lang = max(scores, key=scores.get)

    if scores[best_lang] == 0:
        return "en"

    return best_lang











intent_detector.py
import json
import re

from agents.filters import normalize_query_for_intent
from semantic.semantic_layer import detect_dimension, detect_metric


INTENT_KEYWORDS = {
    "top": ["top", "mas", "mayores"],
    "bottom": ["bottom", "menos", "peores"],
    "average": ["promedio", "average"],
    "total": ["total", "suma"],
}

UNKNOWN_INTENT = {
    "intent": "unknown",
    "metric": None,
    "dimension": None,
    "limit": None,
}


def normalize_intent(intent: dict) -> dict:
    return {
        "intent": intent.get("intent"),
        "metric": intent.get("metric"),
        "dimension": intent.get("dimension"),
        "limit": int(intent.get("limit")) if intent.get("limit") else None,
    }


def detect_fast_intent(query: str) -> str | None:
    for intent, keywords in INTENT_KEYWORDS.items():
        if any(keyword in query for keyword in keywords):
            return intent
    return None


def detect_limit(query: str) -> int | None:
    match = re.search(r"\b(\d+)\b", query)
    return int(match.group(1)) if match else None


def build_llm_prompt(question: str) -> str:
    return f"""
Classify the user intent.

Return ONLY JSON:

{{
  "intent": "top | bottom | average | total | comparison | distribution | unknown",
  "metric": "string or null",
  "dimension": "string or null",
  "limit": "number or null"
}}

Question:
{question}
"""


def detect_llm_intent(question: str, llm) -> dict:
    raw = llm.generate(
        build_llm_prompt(question),
        system_prompt="Intent classifier",
    )

    try:
        return normalize_intent(json.loads(raw))
    except (json.JSONDecodeError, TypeError, ValueError):
        return UNKNOWN_INTENT.copy()


def detect_intent(question: str, llm, semantic_layer=None) -> dict:
    query = normalize_query_for_intent(question)

    intent = detect_fast_intent(query)
    limit = detect_limit(query)

    metric = detect_metric(query, semantic_layer)
    dimension = detect_dimension(query, semantic_layer)

    if intent:
        return normalize_intent(
            {
                "intent": intent,
                "metric": metric,
                "dimension": dimension,
                "limit": limit,
            }
        )

    return detect_llm_intent(question, llm)









relationship_detector.py
MAX_UNIQUE_VALUES = 100

MIN_RELATION_SCORE = 0.8

TEXT_BLACKLIST = {
    "description",
    "comment",
    "text",
    "notes",
    "summary"
}


def normalize_values(values):
    return {
        str(v).strip().lower()
        for v in values
        if str(v).strip()
    }


def relation_score(left_values, right_values):
    """
    Relationship score based on exact overlap.

    1.0 = all values from the smaller set
          exist in the larger set.

    Example:

    {1,2,3}
    {1,2,3,4,5}

    overlap = 3
    score = 3 / min(3,5) = 1.0
    """

    left = normalize_values(left_values)
    right = normalize_values(right_values)

    if not left or not right:
        return 0.0

    overlap = len(left & right)

    denominator = min(
        len(left),
        len(right)
    )

    if denominator == 0:
        return 0.0

    return overlap / denominator


def is_candidate_column(series, column_name):
    """
    Filter obvious text fields and unsuitable columns.
    """

    if column_name.lower() in TEXT_BLACKLIST:
        return False

    if series.dtype != "object":
        return False

    unique_count = series.nunique(dropna=True)

    if unique_count <= 1:
        return False

    if unique_count > 500:
        return False

    avg_len = (
        series.astype(str)
        .str.len()
        .mean()
    )

    return avg_len <= 40


def detect_relationships(datasets):
    """
    Detect relationships between datasets using
    exact value overlap.

    Returns:

    [
        {
            "left_dataset": "...",
            "right_dataset": "...",
            "left_column": "...",
            "right_column": "...",
            "score": 0.95
        }
    ]
    """

    relationships = []

    dataset_names = list(datasets.keys())

    for i in range(len(dataset_names)):

        for j in range(i + 1, len(dataset_names)):

            left_dataset = dataset_names[i]
            right_dataset = dataset_names[j]

            left_df = datasets[left_dataset]["df"]
            right_df = datasets[right_dataset]["df"]

            for lc in left_df.columns:

                if not is_candidate_column(
                    left_df[lc],
                    lc
                ):
                    continue

                left_values = (
                    left_df[lc]
                    .dropna()
                    .astype(str)
                    .unique()[:MAX_UNIQUE_VALUES]
                )

                for rc in right_df.columns:

                    if not is_candidate_column(
                        right_df[rc],
                        rc
                    ):
                        continue

                    right_values = (
                        right_df[rc]
                        .dropna()
                        .astype(str)
                        .unique()[:MAX_UNIQUE_VALUES]
                    )

                    score = relation_score(
                        left_values,
                        right_values
                    )

                    if score >= MIN_RELATION_SCORE:

                        relationships.append({
                            "left_dataset": left_dataset,
                            "right_dataset": right_dataset,
                            "left_column": lc,
                            "right_column": rc,
                            "score": round(score, 2)
                        })

    relationships.sort(
        key=lambda r: r["score"],
        reverse=True
    )

    return relationships

from pipeline.context_stage import build_context
from pipeline.cache_stage import (
    get_cached_response,
    store_response,
)

    run_pre_pipeline,
    run_main_pipeline,
)
from core.observability import new_trace_id
from pipeline.error_handler import ErrorHandler
from core.llm_service import LLMService  # 👈 Fallback por defecto si no nos pasan cliente LLM


def answer_question(
    question: str,
    dataset_ids: list[str] = None,
    dataset_relations: list[dict] = None,
    semantic_layer: dict = None,
    inline_datasets: dict = None,
    llm=None,
    # Parámetros legacy (para no romper Streamlit/Slack antiguos si usan 'connector' o 'dataset_id' único)
    connector=None,
    dataset_id: str = None,
):
    """
    Entry point principal del AI Agent Service.
    Soporta consumo desde FastAPI (CRM/ERP), Streamlit UI y Slack.
    """
    trace_id = new_trace_id()

    # 1. Fallback automático para LLM
    if llm is None:
        llm = LLMService()

    # 2. Normalización de parámetros de entrada (retrocompatibilidad)
    if dataset_ids is None:
        if dataset_id:
            dataset_ids = [dataset_id]
        else:
            dataset_ids = []

    try:
        # 3. Construcción del contexto del pipeline
        ctx = build_context(
            question=question,
            dataset_ids=dataset_ids,
            dataset_relations=dataset_relations,
            semantic_layer=semantic_layer,
            inline_datasets=inline_datasets,
            connector=connector,
            llm=llm,
            trace_id=trace_id,
        )

        # 4. Pre-pipeline (ej. clasificación de intent / normalización)
        ctx = run_pre_pipeline(
            ctx,
            llm,
        )

        # 5. Capa de Cache
        cached = get_cached_response(
            ctx,
            question,
            dataset_ids,
            semantic_layer,
            trace_id,
        )

        if cached:
            return cached

        # 6. Ejecución del Pipeline Principal (SQL generation, execution, insights, charts)
        response = run_main_pipeline(
            ctx,
            llm,
        )

        # 7. Guardado en Cache
        store_response(
            ctx,
            response,
        )

        return response

    except Exception as e:
        return ErrorHandler.to_response(e)

import json


class LLMOutputFormatError(Exception):
    pass


def build_prompt(
    question,
    schema,
    semantic_layer,
    intent=None,
    language="en",
    relationships=None
):

    available_tables = ", ".join(
        schema["table_name"].unique()
    )

    intent_text = ""
    if intent:
        intent_text = f"""
Intent:
Type: {intent.get("intent")}
Metric: {intent.get("metric")}
Dimension: {intent.get("dimension")}
Limit: {intent.get("limit")}
"""

    semantic_text = ""
    if semantic_layer:
        semantic_text = f"""
Semantic Layer:
Aliases: {semantic_layer.get("aliases")}
Metrics: {semantic_layer.get("metrics")}
Columns: {semantic_layer.get("columns")}
"""

    relationships_text = ""
    if relationships:
        relationships_text = f"""
Relationships:
{json.dumps(relationships, indent=2)}

IMPORTANT:
- Use these relationships when JOIN is required.
- Preserve requested entity granularity.
- If relation is one_to_many or many_to_many, aggregate after JOIN.
- Avoid duplicated rows.
- Never duplicate entities due to JOIN.
"""

    return f"""
Return ONLY valid JSON.

Format:
{{
  "sql": "string",
  "confidence": float
}}

Rules:
- Only SELECT
- Use ONLY columns from schema
- No explanation
- No markdown
- Respect intent
- Use JOIN only when necessary

INTENT:
- top -> ORDER BY DESC
- bottom -> ORDER BY ASC
- average -> AVG()
- total -> SUM()

Available tables:
{available_tables}

{intent_text}

Schema:
{schema.to_dict(orient="records")}

{semantic_text}

{relationships_text}

Question:
{question}
"""


def validate_llm_output(parsed):

    if not isinstance(parsed, dict):
        raise LLMOutputFormatError("Output is not a JSON object")

    if "sql" not in parsed:
        raise LLMOutputFormatError("Missing 'sql' field")

    if not isinstance(parsed["sql"], str):
        raise LLMOutputFormatError("SQL must be string")


def generate_sql_structured(
    question,
    schema,
    semantic_layer,
    llm,
    language="en",
    intent=None,
    relationships=None
):

    prompt = build_prompt(
        question=question,
        schema=schema,
        semantic_layer=semantic_layer,
        intent=intent,
        language=language,
        relationships=relationships
    )

    raw = llm.generate(
        prompt,
        system_prompt="You are a senior data analyst. Return ONLY valid JSON."
    )

    if not raw or not raw.strip():
        raise LLMOutputFormatError("Invalid JSON: empty response")

    raw = raw.strip()
    raw = raw.replace("```json", "").replace("```", "").strip()

    try:
        decoder = json.JSONDecoder()
        parsed, _ = decoder.raw_decode(raw)

    except Exception:
        print("LLM RAW RESPONSE:")
        print(raw)
        raise LLMOutputFormatError(f"Invalid JSON: {raw}")

    validate_llm_output(parsed)

    return parsed












sql_guard.py
import sqlglot
from sqlglot import exp

from core.settings import SQLConfig


def guard_sql(
    query: str,
    schema_columns: list[str],
    allowed_tables: list[str] | None = None,
) -> str:
    """
    Validate a SQL query before execution.
    """

    if not query:
        raise ValueError("Empty query")

    try:
        parsed = sqlglot.parse_one(query)
    except Exception as exc:
        raise ValueError("Invalid SQL") from exc

    if not isinstance(parsed, exp.Select):
        raise ValueError("Only SELECT statements are allowed")

    _validate_tables(parsed, allowed_tables)
    _validate_columns(parsed, schema_columns)
    _validate_joins(parsed)

    if not parsed.args.get("limit"):
        parsed.set(
            "limit",
            exp.Limit(
                expression=exp.Literal.number(SQLConfig.MAX_ROWS)
            ),
        )

    return parsed.sql()


def _validate_tables(
    parsed: exp.Expression,
    allowed_tables: list[str] | None,
) -> None:

    if not allowed_tables:
        return

    used_tables = {
        table.name.lower()
        for table in parsed.find_all(exp.Table)
    }

    invalid_tables = (
        used_tables
        - {table.lower() for table in allowed_tables}
    )

    if invalid_tables:
        raise ValueError(
            f"Invalid tables: {sorted(invalid_tables)}"
        )


def _validate_columns(
    parsed: exp.Expression,
    schema_columns: list[str],
) -> None:

    used_columns = {
        column.name
        for column in parsed.find_all(exp.Column)
    }

    invalid_columns = (
        used_columns
        - set(schema_columns)
    )

    if invalid_columns:
        raise ValueError(
            f"Invalid columns: {sorted(invalid_columns)}"
        )


def _validate_joins(
    parsed: exp.Expression,
) -> None:

    joins = list(parsed.find_all(exp.Join))

    if len(joins) > SQLConfig.MAX_JOINS:
        raise ValueError("Too many JOINs")

    if any(
        join.args.get("kind") == "cross"
        for join in joins
    ):
        raise ValueError(
            "CROSS JOIN is not allowed"
        )

    parsed, _ = (
        decoder.raw_decode(raw)
    )

    return parsed

def should_retry(error: str):

    error = error.lower()

    retryable = [
        "invalid json",
        "missing 'sql'",
        "sql inválido",
        "columnas inválidas",
        "tabla inválida",
        "solo select",
        "demasiados joins",
        "expecting value",
        "invalid json: empty response",
    ]

    return any(
        x in error
        for x in retryable
    )


def generate_sql_with_retry(
    question,
    schema,
    semantic_layer,
    columns,
    llm,
    language="en",
    intent=None,
    relationships=None,
    max_retries=2,
):

    last_error = None
    previous_output = None

    for _ in range(
        max_retries + 1
    ):

        try:

            if previous_output is None:

                parsed = (
                    generate_sql_structured(
                        question=question,
                        schema=schema,
                        semantic_layer=semantic_layer,
                        llm=llm,
                        language=language,
                        intent=intent,
                        relationships=relationships,
                    )
                )

            else:

                parsed = (
                    repair_sql_generation(
                        question=question,
                        schema=schema,
                        semantic_layer=semantic_layer,
                        intent=intent,
                        relationships=relationships,
                        previous_output=previous_output,
                        error=last_error,
                        llm=llm,
                    )
                )

            return validate_sql(
                parsed,
                schema,
                columns,
            )

        except Exception as e:

            last_error = str(e)
            previous_output = parsed

            if not should_retry(
                last_error
            ):
                raise

    raise ValueError(
        f"Failed after retries: {last_error}"
    )

from agents.sql_guard import (
    guard_sql,
)


def validate_sql(
    parsed,
    schema,
    columns,
):

    sql = parsed["sql"]

    if "select" not in sql.lower():
        raise ValueError(
            "SQL inválido"
        )

    if len(sql) < 10:
        raise ValueError(
            "SQL demasiado corto"
        )

    tables = (
        schema["table_name"]
        .unique()
        .tolist()
    )

    return guard_sql(
        sql,
        columns,
        allowed_tables=tables,
    )
