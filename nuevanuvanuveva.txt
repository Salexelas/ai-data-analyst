def is_valid_question(question: str) -> bool:
    """Validación mínima de longitud e intención de texto."""
    q = question.strip()
    return len(q) >= 3 and any(c.isalpha() for c in q)

import json


class LLMOutputFormatError(Exception):
    pass


def format_examples(examples: list) -> str:
    if not examples:
        return ""
    lines = ["EJEMPLOS DE REFERENCIA (Few-Shot):"]
    for ex in examples:
        lines.append(f"Pregunta: {ex.get('question')}\nSQL: {ex.get('sql', '').strip()}\n---")
    return "\n".join(lines)


def format_taxonomies(taxonomies: dict) -> str:
    if not taxonomies:
        return ""
    return f"TAXONOMIAS Y VALORES CANONICOS:\n{json.dumps(taxonomies, indent=2, ensure_ascii=False)}"


def build_sql_prompt(
    question: str,
    semantic_layer: dict,
    taxonomies: dict = None,
    examples: list = None,
    intent: dict = None,
) -> str:
    
    examples_text = format_examples(examples or [])
    taxonomies_text = format_taxonomies(taxonomies or {})
    intent_text = f"INTENT DETECTADO: {json.dumps(intent)}\n" if intent else ""

    # Extraemos solo métricas y columnas para mantener el prompt compacto
    compact_semantic = {
        "dataset_id": semantic_layer.get("dataset_id"),
        "columns": list(semantic_layer.get("columns", {}).keys()),
        "metrics": semantic_layer.get("metrics", {}),
        "aliases": semantic_layer.get("aliases", {}),
    }

    return f"""
Eres un analista de datos experto en SQL. Genera la consulta SQL para responder a la pregunta.

METADATOS DEL DATASET:
{json.dumps(compact_semantic, indent=2, ensure_ascii=False)}

{taxonomies_text}

{examples_text}

{intent_text}

REGLAS DE ORO:
1. Responde UNICAMENTE con un JSON válido: {{"sql": "SELECT ...", "confidence": 0.95}}
2. Usa SOLO sentencias SELECT.
3. Respeta las funciones y lógica mostrada en los ejemplos Few-Shot (ej. JULIANDAY, REPLACE, etc.).
4. No agregues texto ni formato Markdown fuera del JSON.

Pregunta del usuario:
"{question}"
"""


def generate_sql_structured(
    question: str,
    semantic_layer: dict,
    llm,
    taxonomies: dict = None,
    examples: list = None,
    intent: dict = None,
) -> str:  # Retorna str
    
    if isinstance(llm, dict):
        raise TypeError(f"Se esperaba LLMService, se recibió dict: {llm}")

    prompt = build_sql_prompt(
        question=question,
        semantic_layer=semantic_layer,
        taxonomies=taxonomies,
        examples=examples,
        intent=intent,
    )

    raw = llm.generate(
        prompt,
        system_prompt="You are a senior data analyst. Return ONLY valid JSON.",
    )

    if not raw or not str(raw).strip():
        raise LLMOutputFormatError("Respuesta vacía del LLM")

    raw = str(raw).strip().replace("```json", "").replace("```", "").strip()

    try:
        parsed = json.loads(raw)
    except Exception as e:
        raise LLMOutputFormatError(f"Error al decodificar JSON: {raw}") from e

    if "sql" not in parsed or not isinstance(parsed["sql"], str):
        raise LLMOutputFormatError("Falta el campo 'sql' en el JSON devuelto")

    # 🔑 CORRECCIÓN: Retornar solo el string de la consulta SQL
    return parsed["sql"].strip()

import sqlglot
from sqlglot import exp
from core.settings import SQLConfig


def guard_sql(
    query: str,
    schema_columns: list[str] | list[dict] | dict,
    allowed_tables: list[str] | list[dict] | None = None,
) -> str:
    """Validate a SQL query before execution."""

    if not query:
        raise ValueError("Empty query")

    try:
        parsed = sqlglot.parse_one(query)
    except Exception as exc:
        raise ValueError("Invalid SQL") from exc

    if not isinstance(parsed, exp.Select):
        raise ValueError("Only SELECT statements are allowed")

    # Normalizar entradas si provienen directamente del YAML
    norm_tables = _normalize_to_list(allowed_tables)
    norm_columns = _normalize_to_list(schema_columns)

    _validate_tables(parsed, norm_tables)
    _validate_columns(parsed, norm_columns)
    _validate_joins(parsed)

    if not parsed.args.get("limit"):
        parsed.set(
            "limit",
            exp.Limit(
                expression=exp.Literal.number(SQLConfig.MAX_ROWS)
            ),
        )

    return parsed.sql()


def _normalize_to_list(data) -> list[str]:
    """Soporta listas de strings, listas de diccionarios YAML o claves de dicts."""
    if not data:
        return []
    if isinstance(data, dict):
        # Si es un dict tipo {'columns': ['a', 'b']} o {'col1': {...}}
        return list(data.keys())
    if isinstance(data, list):
        result = []
        for item in data:
            if isinstance(item, str):
                result.append(item)
            elif isinstance(item, dict):
                # Si en el YAML la columna es {'name': 'producto', 'type': 'str'}
                val = item.get("name") or item.get("column") or item.get("id")
                if val:
                    result.append(str(val))
        return result
    return []


def _validate_tables(
    parsed: exp.Expression,
    allowed_tables: list[str],
) -> None:
    if not allowed_tables:
        return

    used_tables = {
        table.name.lower()
        for table in parsed.find_all(exp.Table)
    }

    invalid_tables = (
        used_tables
        - {table.lower() for table in allowed_tables}
    )

    if invalid_tables:
        raise ValueError(
            f"Invalid tables: {sorted(invalid_tables)}"
        )


def _validate_columns(
    parsed: exp.Expression,
    schema_columns: list[str],
) -> None:
    if not schema_columns:
        return

    physical_columns = {
        str(column).lower()
        for column in schema_columns
    }

    # CTEs disponibles en el query.
    cte_scopes = _collect_cte_scopes(parsed)

    # Derived tables:
    # FROM (...) r
    derived_scopes = _collect_derived_scopes(parsed)

    invalid_columns = set()

    for column in parsed.find_all(exp.Column):
        name = column.name.lower()
        table = column.table.lower() if column.table else None

        # ---------------------------------------------------------
        # 1. Physical column
        # ---------------------------------------------------------
        if name in physical_columns:
            continue

        # ---------------------------------------------------------
        # 2. Column exposed by a derived table / CTE
        # ---------------------------------------------------------
        if table:
            if table in derived_scopes:
                if name in derived_scopes[table]:
                    continue

            if table in cte_scopes:
                if name in cte_scopes[table]:
                    continue

        # ---------------------------------------------------------
        # 3. SELECT alias
        # ---------------------------------------------------------
        if _is_valid_select_alias(column, parsed):
            continue

        invalid_columns.add(name)

    if invalid_columns:
        raise ValueError(
            f"Invalid columns: {sorted(invalid_columns)}"
        )


def _collect_derived_scopes(
    parsed: exp.Expression,
) -> dict[str, set[str]]:
    """
    Detecta:

        FROM (
            SELECT ...
        ) r

    y construye:

        {
            "r": {
                "Responsable",
                "cnt"
            }
        }
    """

    scopes = {}

    for subquery in parsed.find_all(exp.Subquery):
        alias = subquery.alias

        if not alias:
            continue

        inner = subquery.this

        if not isinstance(inner, exp.Select):
            continue

        output_columns = _get_select_output_columns(inner)

        scopes[alias.lower()] = output_columns

    return scopes


def _collect_cte_scopes(
    parsed: exp.Expression,
) -> dict[str, set[str]]:
    """
    Detecta:

        WITH counts AS (
            SELECT ...
        )

    y construye:

        {
            "counts": {
                "Responsable",
                "cnt"
            }
        }
    """

    scopes = {}

    for cte in parsed.find_all(exp.CTE):
        alias = cte.alias

        if not alias:
            continue

        inner = cte.this

        if not isinstance(inner, exp.Select):
            continue

        output_columns = _get_select_output_columns(inner)

        scopes[alias.lower()] = output_columns

    return scopes


def _get_select_output_columns(
    select: exp.Select,
) -> set[str]:
    """
    Devuelve los nombres de las columnas que expone
    un SELECT hacia su scope exterior.

    Ejemplo:

        SELECT
            "Responsable",
            COUNT(*) AS cnt

    devuelve:

        {
            "responsable",
            "cnt"
        }
    """

    output_columns = set()

    for expression in select.expressions:

        # SELECT ... AS alias
        if expression.alias:
            output_columns.add(
                expression.alias.lower()
            )
            continue

        # SELECT "Responsable"
        if isinstance(expression, exp.Column):
            output_columns.add(
                expression.name.lower()
            )
            continue

        # Otros casos donde SQLGlot puede conocer el nombre.
        name = expression.name

        if name:
            output_columns.add(
                name.lower()
            )

    return output_columns


def _is_valid_select_alias(
    column: exp.Column,
    parsed: exp.Expression,
) -> bool:
    """
    Permite aliases generados por el SELECT principal.

    Ejemplo:

        SELECT COUNT(*) AS diferencia
        FROM erp_tasks
        ORDER BY diferencia DESC

    'diferencia' no es una columna física,
    pero sí es un alias válido del SELECT.
    """

    if column.table:
        return False

    name = column.name.lower()

    # Buscar SELECTs que expongan ese alias.
    for select in parsed.find_all(exp.Select):
        for expression in select.expressions:
            if expression.alias:
                if expression.alias.lower() == name:
                    return True

    return False


def _validate_joins(
    parsed: exp.Expression,
) -> None:
    joins = list(parsed.find_all(exp.Join))

    if len(joins) > SQLConfig.MAX_JOINS:
        raise ValueError("Too many JOINs")

    if any(
        join.args.get("kind") == "cross"
        for join in joins
    ):
        raise ValueError("CROSS JOIN is not allowed")

def repair_sql_generation(
    question,
    schema,
    semantic_layer,
    intent,
    previous_output,
    error,
    llm,
):
    fix_prompt = f"""
You generated an invalid SQL query.

Fix it using the error.

Context:

Question:
{question}

Schema:
{schema}

Semantic Layer:
{semantic_layer}

Intent:
{intent}

Previous output:
{previous_output}

Error:
{error}

Rules:
- Keep the same intent
- Preserve entity granularity
- Do not invent columns
- Do not invent tables
- Fix ONLY the error
- Use only the provided schema and semantic layer
- Return ONLY valid JSON

Format:
{{
  "sql": "SELECT ...",
  "confidence": 0.0-1.0
}}
"""

    raw = llm.generate(
        fix_prompt,
        system_prompt="You fix SQL queries",
    )

    if not raw or not raw.strip():
        raise ValueError("Invalid JSON")

    raw = (
        raw.strip()
        .replace("```json", "")
        .replace("```", "")
        .strip()
    )

    decoder = json.JSONDecoder()

    parsed, _ = decoder.raw_decode(raw)

    return parsed

def should_retry(error: str) -> bool:
    error_lower = error.lower()

    retryable_patterns = [
        "json",
        "missing 'sql'",
        "invalid sql",
        "column",
        "table",
        "select",
    ]

    return any(
        pattern in error_lower
        for pattern in retryable_patterns
    )


def generate_sql_with_retry(
    question: str,
    schema_columns: list[str],
    semantic_layer: dict,
    llm,
    taxonomies: dict = None,
    examples: list = None,
    intent: dict = None,
    max_retries: int = 2,
) -> str:

    last_error = None
    previous_output = None

    # ---------------------------------------------------------
    # Tabla permitida
    # ---------------------------------------------------------

    dataset_id = (
        semantic_layer.get("dataset_id")
        if isinstance(semantic_layer, dict)
        else None
    )

    allowed_tables = (
        [dataset_id]
        if dataset_id
        else None
    )

    # ---------------------------------------------------------
    # Retry loop
    # ---------------------------------------------------------

    for attempt in range(max_retries + 1):

        parsed = None

        try:

            # =================================================
            # Primera generación
            # =================================================

            if previous_output is None:

                parsed = generate_sql_structured(
                    question=question,
                    semantic_layer=semantic_layer,
                    llm=llm,
                    taxonomies=taxonomies,
                    examples=examples,
                    intent=intent,
                )
                print("\n" + "=" * 80)
                print("SQL GENERATION DEBUG")
                print("=" * 80)
                print("QUESTION:")
                print(question)
                print("\nPARSED OUTPUT:")
                print(parsed)
                print("=" * 80 + "\n")

            # =================================================
            # Reparación
            # =================================================

            else:

                parsed = repair_sql_generation(
                    question=question,
                    schema=schema_columns,
                    semantic_layer=semantic_layer,
                    intent=intent,
                    previous_output=previous_output,
                    error=last_error,
                    llm=llm,
                )

            # =================================================
            # Validación
            # =================================================

            return validate_sql(
                parsed=parsed,
                schema_columns=schema_columns,
                allowed_tables=allowed_tables,
            )

        except Exception as e:

            last_error = str(e)

            # Guardar la salida anterior si existe
            if parsed is not None and parsed is not ...:
                previous_output = parsed

            # Si el error no es recuperable, salir inmediatamente
            if not should_retry(last_error):
                raise

    raise ValueError(
        f"Fallo tras {max_retries} reintentos. "
        f"Último error: {last_error}"
    )

from agents.sql_guard import guard_sql

def validate_sql(
    parsed: str | dict,
    schema_columns: list[str],
    allowed_tables: list[str] = None
) -> str:
    # 1. Proteger contra Ellipsis (...) o None
    if parsed is ... or parsed is None:
        raise ValueError("SQL inválido: No se recibió ninguna consulta (recibido Ellipsis/None).")

    # 2. Extraer el string SQL según el tipo
    if isinstance(parsed, dict):
        sql = parsed.get("sql", "")
    elif isinstance(parsed, str):
        sql = parsed
    else:
        sql = str(parsed)

    # 3. Validar que sea un string válido y no esté vacío
    sql_clean = sql.strip()

    if not sql_clean or "select" not in sql_clean.lower():
        raise ValueError("SQL inválido: Debe contener una instrucción SELECT.")

    if len(sql_clean) < 10:
        raise ValueError("SQL demasiado corto.")

    # 4. Guard de seguridad en el SQL
    return guard_sql(
        sql_clean,
        schema_columns,
        allowed_tables=allowed_tables,
    )

def sanitize_for_json(obj):
    """
    Convierte dinámicamente cualquier tipo de NumPy/Pandas o DataFrames 
    a tipos nativos de Python serializables en JSON.
    """
    if isinstance(obj, pd.DataFrame):
        return sanitize_for_json(obj.to_dict(orient="records"))
    elif isinstance(obj, pd.Series):
        return sanitize_for_json(obj.tolist())
    elif isinstance(obj, dict):
        return {k: sanitize_for_json(v) for k, v in obj.items()}
    elif isinstance(obj, (list, tuple)):
        return [sanitize_for_json(v) for v in obj]
    elif isinstance(obj, (np.integer, np.int64, np.int32)):
        return int(obj)
    elif isinstance(obj, (np.floating, np.float64, np.float32)):
        return None if np.isnan(obj) else float(obj)
    elif isinstance(obj, (np.bool_, bool)):
        return bool(obj)
    elif pd.isna(obj) if isinstance(obj, (float, int, str, np.number)) else False:
        return None
    elif hasattr(obj, "item") and callable(getattr(obj, "item")):
        return obj.item()
    return obj

from typing import Any, Optional
from core.dataset_store import DatasetStore


def build_context(
    question: str,
    dataset_id: str = None,
    dataset_ids: list[str] = None,
    dataset_relations: list[dict] = None,
    semantic_layer: dict = None,
    taxonomies: dict = None,
    examples: list = None,
    inline_datasets: dict = None,
    connector=None,
    llm=None,
    trace_id: str = None,
    **kwargs,
) -> dict:
    """Construye el contexto.

    Registra el YAML la primera vez y lo reutiliza de memoria en subsecuentes
    preguntas.
    """
    target_id = (
        dataset_id
        or (dataset_ids[0] if dataset_ids else None)
        or (semantic_layer.get("dataset_id") if semantic_layer else None)
    )

    # 1. Si enviaron el YAML completo en este request, registrarlo / actualizarlo en memoria
    if semantic_layer and "columns" in semantic_layer:
        d_id = semantic_layer.get("dataset_id", target_id)
        if d_id:
            DatasetStore.register(d_id, semantic_layer)
            target_id = d_id

    # 2. Si solo mandaron el ID (preguntas subsecuentes), recuperarlo del store
    elif target_id and DatasetStore.exists(target_id):
        semantic_layer = DatasetStore.get(target_id)

    else:
        semantic_layer = semantic_layer or {}

    # 3. Prioridad a los objetos enviados explícitamente, de lo contrario se leen de semantic_layer
    resolved_taxonomies = taxonomies or semantic_layer.get("taxonomies", {})
    resolved_examples = examples or semantic_layer.get("examples", [])

    return {
        "question": question,
        "trace_id": trace_id,
        "dataset_id": target_id,
        "dataset_ids": dataset_ids or ([target_id] if target_id else []),
        "dataset_relations": dataset_relations or [],
        "inline_datasets": inline_datasets or {},

        # NUEVO: conservar el schema del dataset
        "inline_dataset_schema": kwargs.get(
            "inline_dataset_schema",
            {},
        ),

        "llm": llm,
        "connector": connector,
        "semantic_layer": semantic_layer,
        "taxonomies": resolved_taxonomies,
        "examples": resolved_examples,
        "schema_columns": list(
            semantic_layer.get("columns", {}).keys()
        ),
    }

def run_pre_pipeline(ctx: dict, llm=None) -> dict:
    """Validación previa rápida de la entrada sin llamadas al LLM."""

def _restore_date_columns(df: pd.DataFrame) -> pd.DataFrame:
    result = df.copy()

    for column in result.columns:
        series = result[column]

        if series.dtype != "object":
            continue

        values = series.dropna()

        if values.empty:
            continue

        converted = pd.to_datetime(
            values,
            errors="coerce",
        )

        if converted.notna().all():
            result[column] = pd.to_datetime(
                series,
                errors="coerce",
            )

    return result


def _execute_inline_sql(
    sql: str,
    inline_datasets: dict,
    semantic_layer: dict,
    inline_dataset_schema: dict = None,
) -> pd.DataFrame:
    """
    Ejecuta el SQL generado por el Agent sobre los datasets
    enviados por la aplicación consumidora.

    Reconstruye los tipos originales del DataFrame antes
    de registrarlo en DuckDB.
    """

    if not sql:
        raise ValueError(
            "No se recibió una consulta SQL."
        )

    if not inline_datasets:
        raise ValueError(
            "No se recibieron datasets inline."
        )

    dataset_id = semantic_layer.get("dataset_id")

    if not dataset_id:
        raise ValueError(
            "El semantic layer no contiene 'dataset_id'."
        )

    records = inline_datasets.get(dataset_id)

    if records is None and len(inline_datasets) == 1:
        records = next(iter(inline_datasets.values()))

    if records is None:
        raise ValueError(
            f"No se encontró el dataset '{dataset_id}' "
            "en inline_datasets."
        )

    # ---------------------------------------------------------
    # Convertir records a DataFrame
    # ---------------------------------------------------------

    if isinstance(records, pd.DataFrame):

        df = records.copy()

    elif isinstance(records, list):

        df = pd.DataFrame(records)


    else:

        raise TypeError(
            f"Formato no soportado para inline_datasets['{dataset_id}']: "
            f"{type(records).__name__}"
        )

    df = _restore_date_columns(df)
    
    # ---------------------------------------------------------
    # Restaurar tipos originales
    # ---------------------------------------------------------

    dataset_schema = (
        (inline_dataset_schema or {})
        .get(dataset_id)
    )

    if dataset_schema:

        for column, dtype in dataset_schema.items():

            if column not in df.columns:
                continue

            if dtype.startswith("datetime64"):

                df[column] = pd.to_datetime(
                    df[column],
                    errors="coerce",
                )

            elif dtype == "date":

                df[column] = pd.to_datetime(
                    df[column],
                    errors="coerce",
                ).dt.date

            elif dtype.startswith("int"):

                df[column] = pd.to_numeric(
                    df[column],
                    errors="coerce",
                ).astype("Int64")

            elif dtype.startswith("float"):

                df[column] = pd.to_numeric(
                    df[column],
                    errors="coerce",
                ).astype("Float64")

            elif dtype in ("bool", "boolean"):

                df[column] = df[column].astype("boolean")

            elif dtype in ("string", "object"):

                df[column] = df[column].astype("string")

    # ---------------------------------------------------------
    # Ejecutar SQL
    # ---------------------------------------------------------

    con = duckdb.connect()

    try:

        con.register(
            dataset_id,
            df,
        )

        result_df = con.execute(sql).df()

        return result_df

    finally:

        con.close()


def run_main_pipeline(ctx: dict, llm) -> dict:

    # =========================================================
    # 1. GENERAR SQL
    # =========================================================

    sql_query = generate_sql_with_retry(
        question=ctx["question"],
        schema_columns=ctx["schema_columns"],
        semantic_layer=ctx["semantic_layer"],
        llm=llm,
        taxonomies=ctx.get("taxonomies"),
        examples=ctx.get("examples"),
        intent=ctx.get("intent"),
    )

    if isinstance(sql_query, dict):
        sql_query = sql_query.get("sql", "")

    if not sql_query:
        raise ValueError(
            "El Agent no generó una consulta SQL válida."
        )

    ctx["generated_sql"] = sql_query

    # =========================================================
    # 2. EJECUTAR SQL SOBRE LOS DATOS REALES
    # =========================================================

    result_df = _execute_inline_sql(
        sql=sql_query,
        inline_datasets=ctx.get("inline_datasets"),
        semantic_layer=ctx.get("semantic_layer"),
        inline_dataset_schema=ctx.get("inline_dataset_schema"),
    )

    # =========================================================
    # 3. CONVERTIR RESULTADOS A JSON-SERIALIZABLE
    # =========================================================

    result_data = result_df.to_dict(
        orient="records"
    )

    # =========================================================
    # 4. SEGUNDO LLM
    #
    # IMPORTANTE:
    # Ahora el LLM recibe LOS RESULTADOS.
    # =========================================================

    answer_prompt = f"""
Eres un asistente de datos.

Pregunta del usuario:
{ctx["question"]}

Resultados reales obtenidos de la consulta:
{result_data}

Responde directamente la pregunta del usuario utilizando
únicamente los resultados proporcionados.

No inventes información.

No expliques el SQL salvo que sea necesario.

No digas que el usuario debe ejecutar la consulta.

Si hay resultados, responde con la información encontrada.

Si no hay resultados, indícalo claramente.
"""

    answer = llm.generate(
        answer_prompt,
        system_prompt="You are a helpful data analyst.",
    )

    # =========================================================
    # 5. RESPUESTA FINAL
    # =========================================================

    return {
        "status": "success",
        "trace_id": ctx.get("trace_id"),
        "question": ctx["question"],
        "sql": sql_query,
        "data": result_data,
        "answer": answer,
    }

import json
from typing import Any, Optional
from datetime import date, datetime
import pandas as pd
import requests


def consultar_agente(
    pregunta: str,
    dataset_name: str,
    df: pd.DataFrame,
    semantic_layer_yaml: Optional[dict[str, Any]] = None,
    taxonomies_yaml: Optional[dict[str, Any]] = None,
    examples_yaml: Optional[list[dict[str, Any]] | dict[str, Any]] = None,
) -> dict:

    url = "http://localhost:8000/query"

    # 1. Conversión del DataFrame
    records = json.loads(
        df.to_json(
            orient="records",
            date_format="iso",
        )
    )

    # 2. Preservar los tipos originales del DataFrame
    def detectar_tipo_columna(series: pd.Series) -> str:
        dtype = series.dtype

        if pd.api.types.is_datetime64_any_dtype(dtype):
            return "datetime64"

        if pd.api.types.is_integer_dtype(dtype):
            return "int"

        if pd.api.types.is_float_dtype(dtype):
            return "float"

        if pd.api.types.is_bool_dtype(dtype):
            return "bool"

        if dtype == "object":
            valores = series.dropna()

            if not valores.empty:
                valor = valores.iloc[0]

                if isinstance(valor, datetime):
                    return "datetime64"

                if isinstance(valor, date):
                    return "date"

            return "object"

        return str(dtype)


    schema = {
        column: detectar_tipo_columna(df[column])
        for column in df.columns
    }

    # 3. Normalización de examples
    examples_list = examples_yaml or []

    if isinstance(examples_list, dict) and "examples" in examples_list:
        examples_list = examples_list["examples"]

    # 4. Payload
    payload = {
        "question": pregunta,
        "dataset_ids": [dataset_name],
        "dataset_relations": [],
        "inline_datasets": {
            dataset_name: records,
        },
        "inline_dataset_schema": {
            dataset_name: schema,
        },
        "semantic_layer": semantic_layer_yaml or {},
        "taxonomies": taxonomies_yaml or {},
        "examples": examples_list,
    }

    try:
        response = requests.post(
            url,
            json=payload,
            timeout=60,
        )

        response.raise_for_status()

        return response.json()

    except Exception as e:

        return {
            "error": f"Error conectando con el AI Agent: {e}"
        }

import pandas as pd
import streamlit as st
import yaml

from app.services.ai_agent_client import consultar_agente


def render_asistente_ia_dataset(
    dataset_name: str,
    df: pd.DataFrame,
):
    """
    Renderiza un componente genérico de consulta IA
    para cualquier dataset.
    """

    st.markdown("---")

    st.subheader(
        f"🤖 Consultar al AI Agent ({dataset_name})"
    )

    if df.empty:
        st.info(
            "No hay datos cargados en esta vista para analizar."
        )
        return

    # =========================================================
    # SESSION STATE
    # =========================================================

    yaml_state_key = f"yaml_config_{dataset_name}"
    input_key = f"input_ia_{dataset_name}"
    btn_key = f"btn_ia_{dataset_name}"
    state_key = f"res_ia_{dataset_name}"

    if yaml_state_key not in st.session_state:

        st.session_state[yaml_state_key] = {
            "semantic": {},
            "taxonomies": {},
            "examples": [],
        }

    # =========================================================
    # YAML CONFIGURATION
    # =========================================================

    with st.expander(
        f"⚙️ Configurar YAMLs para {dataset_name} "
        "(opcional / se carga una sola vez)"
    ):

        col1, col2, col3 = st.columns(3)

        # -----------------------------------------------------
        # Semantic layer
        # -----------------------------------------------------

        with col1:

            up_sem = st.file_uploader(
                "Subir semantic_layer.yaml",
                type=["yaml", "yml"],
                key=f"up_sem_{dataset_name}",
            )

            if up_sem:

                st.session_state[yaml_state_key]["semantic"] = (
                    yaml.safe_load(up_sem) or {}
                )

        # -----------------------------------------------------
        # Taxonomies
        # -----------------------------------------------------

        with col2:

            up_tax = st.file_uploader(
                "Subir taxonomies.yaml",
                type=["yaml", "yml"],
                key=f"up_tax_{dataset_name}",
            )

            if up_tax:

                st.session_state[yaml_state_key]["taxonomies"] = (
                    yaml.safe_load(up_tax) or {}
                )

        # -----------------------------------------------------
        # Examples
        # -----------------------------------------------------

        with col3:

            up_ex = st.file_uploader(
                "Subir query_examples.yaml",
                type=["yaml", "yml"],
                key=f"up_ex_{dataset_name}",
            )

            if up_ex:

                data_ex = yaml.safe_load(up_ex) or []

                if (
                    isinstance(data_ex, dict)
                    and "examples" in data_ex
                ):
                    data_ex = data_ex["examples"]

                st.session_state[yaml_state_key]["examples"] = (
                    data_ex
                )

        config = st.session_state[yaml_state_key]

        st.caption(
            f"**Estado:** "
            f"Semantic "
            f"{'✅' if config['semantic'] else '❌'} | "
            f"Taxonomías "
            f"{'✅' if config['taxonomies'] else '❌'} | "
            f"Ejemplos "
            f"{'✅' if config['examples'] else '❌'}"
        )

    # =========================================================
    # QUESTION
    # =========================================================

    pregunta = st.text_input(
        f"Preguntale algo a la IA sobre {dataset_name}:",
        placeholder=(
            "Ej: ¿Cuáles son las tareas completas de Tyler?"
        ),
        key=input_key,
    )

    # =========================================================
    # CALL AGENT
    # =========================================================

    if st.button(
        "Analizar con IA",
        key=btn_key,
    ):

        if not pregunta.strip():

            st.warning(
                "Por favor, ingresá una pregunta."
            )

            return

        config_dataset = (
            st.session_state[yaml_state_key]
        )

        with st.spinner(
            "El AI Agent está procesando los datos..."
        ):

            resultado = consultar_agente(
                pregunta=pregunta,
                dataset_name=dataset_name,
                df=df,
                semantic_layer_yaml=config_dataset["semantic"],
                taxonomies_yaml=config_dataset["taxonomies"],
                examples_yaml=config_dataset["examples"],
            )

            st.session_state[state_key] = resultado

    # =========================================================
    # SHOW RESULT
    # =========================================================

    if state_key not in st.session_state:
        return

    resultado = st.session_state[state_key]

    # =========================================================
    # ERROR
    # =========================================================

    if resultado.get("error"):

        st.error(
            resultado["error"]
        )

        return

    # =========================================================
    # ANSWER
    # =========================================================

    answer = resultado.get("answer")

    if answer:

        st.markdown(
            "### 🤖 Respuesta del AI Agent"
        )

        st.markdown(answer)

    # =========================================================
    # DATA
    # =========================================================

    data = resultado.get("data")

    if data:

        df_resultado = pd.DataFrame(data)

        st.markdown(
            "### 📊 Resultado"
        )

        st.dataframe(
            df_resultado,
            use_container_width=True,
            hide_index=True,
        )

        st.caption(
            f"{len(df_resultado)} resultado(s)"
        )

    elif "data" in resultado:

        st.info(
            "La consulta no devolvió resultados."
        )

    # =========================================================
    # SQL
    # =========================================================

    sql = resultado.get("sql")

    if sql:

        with st.expander(
            "🔍 Ver SQL generado por el Agente"
        ):

            st.code(
                sql,
                language="sql",
            )
