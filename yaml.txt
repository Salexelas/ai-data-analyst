3. query_examples.yaml
YAML
dataset_id: erp_tasks
examples:
  - question: "¿Cuántas tareas terminadas tiene cada responsable?"
    pattern_type: taxonomy_grouping
    sql: >
      SELECT "Responsable", COUNT(*) AS tareas_terminadas 
      FROM erp_tasks 
      WHERE "Estado" IN ('Terminado', 'Completada', 'Finalizada', 'Finalizado', 'OK') 
      GROUP BY "Responsable" 
      ORDER BY tareas_terminadas DESC;

  - question: "¿Cuáles son las tareas cuya duración real fue mayor a la estimada?"
    pattern_type: date_difference_comparison
    sql: >
      SELECT "Tarea", "Responsable", 
             (JULIANDAY("Fin") - JULIANDAY("Inicio")) AS dias_reales, 
             CAST(REPLACE("Duración estimada", ' días', '') AS INTEGER) AS dias_estimados 
      FROM erp_tasks 
      WHERE (JULIANDAY("Fin") - JULIANDAY("Inicio")) > CAST(REPLACE("Duración estimada", ' días', '') AS INTEGER);

  - question: "¿Qué tareas del proyecto GN103AJ se ejecutaron en la sede hoffner?"
    pattern_type: multi_filter
    sql: >
      SELECT "Tarea", "Estado", "Inicio", "Fin" 
      FROM erp_tasks 
      WHERE "Proyecto" = 'GN103AJ' 
        AND LOWER("Fábrica/Sede") = 'hoffner';
